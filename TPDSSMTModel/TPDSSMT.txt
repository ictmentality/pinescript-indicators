f_live_sym(_xc0,_xo0,_xh0,_xl0,_xt0,_tf)=>
    var _c1 = 0.0, var _o1 = 0.0 , var _h1 = 0.0, var _l1 = 0.0, var _t1 = 0
    var _c2 = 0.0, var _o2 = 0.0 , var _h2 = 0.0, var _l2 = 0.0, var _t2 = 0
    var up0 = false , var up1 = false
    var dw0 = false , var dw1 = false
    var _l0t = time , var _h0t = time, var _h1t = time , var _l1t = time
    var _o0 = open
    var _h0 = high
    var _l0 = low
    var _t0 = time
    var _c0 = close
    if timeframe.change(_tf)
        _t0 := _xt0
        _o0 := _xo0
        _h0 := _xh0
        _l0 := _xl0

    if _xh0 >= _h0
        _h0 := _xh0
        _h0t := time
    if _xl0 <= _l0
        _l0 := _xl0
        _l0t := time
    _c0 := _xc0
    up0 := _c0 > _o0
    dw0 := _c0 < _o0
    if timeframe.change(_tf)
        _c1  :=  _c0[1]
        _o1  :=  _o0[1]
        _h1  :=  _h0[1]
        _l1  :=  _l0[1]
        _t1  :=  _t0[1]
        _c2  :=  _c1[1]
        _o2  :=  _o1[1]
        _h2  :=  _h1[1]
        _l2  :=  _l1[1]
        _t2  :=  _t1[1]
        up1  :=  up0[1]
        dw1  :=  dw0[1]
        _l1t :=  _l0t[1]
        _h1t :=  _h0t[1]

    [_h2,_l2,_c2,_o2,_t2,_h1,_l1,_c1,_o1,_t1,_h0,_l0,_c0,_o0,_t0,up1,dw1,_l1t,_h1t]

f_qt_pick_qr(int idx, qt.QuarterRecord q1, qt.QuarterRecord q2, qt.QuarterRecord q3, qt.QuarterRecord q4) =>
    qt.QuarterRecord out = q1
    if idx == 2
        out := q2
    else if idx == 3
        out := q3
    else if idx == 4
        out := q4
    out

f_qt_daily_c123() =>
    var qt.QTConfig cfg = qt.qt_config_default()
    var qt.CycleState st = qt.qt_state_new()
    [st0, d] = qt.qt_daily_update(st, cfg, time, time_close)
    st := st0

    qt.QuarterRecord qcur = f_qt_pick_qr(d.curQuarterIndex, d.q1, d.q2, d.q3, d.q4)
    qt.QuarterRecord qprev = d.curQuarterIndex == 1 ? st.daily.prev1 : d.curQuarterIndex == 2 ? d.q1 : d.curQuarterIndex == 3 ? d.q2 : d.q3
    qt.QuarterRecord qprev2 = d.curQuarterIndex == 1 ? st.daily.prev2 : d.curQuarterIndex == 2 ? st.daily.prev1 : d.curQuarterIndex == 3 ? d.q1 : d.q2
    bool qprev_up = qprev.has and not na(qprev.c) and not na(qprev.o) and qprev.c > qprev.o
    bool qprev_down = qprev.has and not na(qprev.c) and not na(qprev.o) and qprev.c < qprev.o

    [qprev2.h, qprev2.l, qprev2.c, qprev2.o, qprev2.boundaryTs, qprev.h, qprev.l, qprev.c, qprev.o, qprev.boundaryTs, qcur.h, qcur.l, qcur.c, qcur.o, qcur.boundaryTs, qprev_up, qprev_down, qprev.lTs, qprev.hTs]

f_main_c123(_htf) =>
    if timeframe.in_seconds(_htf) == 6 * 60 * 60
        request.security(syminfo.type == "futures" ? ticker.modify(sym, session = syminfo.session, backadjustment=badj ? backadjustment.on:backadjustment.off) : sym, "15", f_qt_daily_c123(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off, ignore_invalid_symbol=true)
    else
        f_live_sym(close, open, high, low, time, _htf)

f_pair_c123(_htf, _csym, _xc0, _xo0, _xh0, _xl0, _xt0) =>
    if timeframe.in_seconds(_htf) == 6 * 60 * 60
        request.security(_csym, "15", f_qt_daily_c123(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off, ignore_invalid_symbol=true)
    else
        f_live_sym(_xc0, _xo0, _xh0, _xl0, _xt0, _htf)

f_text(string _tf)=>
    float _tf_num = str.tonumber(_tf)
    int _tf_int = int(na)
    string _text = _tf
    if not na(_tf_num)
        _tf_int := int(_tf_num)
    if _tf == "1D"
        _text := "D"
    else if _tf == "1W"
        _text := "W"
    else if _tf == "1M"
        _text := "M"
    else if not na(_tf_int)
        if _tf_int < 60
            _text := "m" + _tf
        else if _tf_int % 60 == 0
            _text := "H" + str.tostring(_tf_int / 60)
        else
            _text := "m" + _tf
    _text

f_short_sym(string _s)=>
    string out = _s
    if str.length(out) == 0
        out := ""
    else if str.contains(out, "replay-feed")
        out := syminfo.ticker
    else if str.contains(out, ":")
        out := array.get(str.split(out, ":"), 1)
    out

f_fmt(float _v)=>

f_fmt_time(int _t)=>

f_smt_pair(_lo_c2_a, _lo_c3_a, _hi_c2_a, _hi_c3_a, _lo_c2_b, _lo_c3_b, _hi_c2_b, _hi_c3_b)=>
    bool a_low_sweep = not na(_lo_c2_a) and not na(_lo_c3_a) and _lo_c3_a < _lo_c2_a
    bool b_low_sweep = not na(_lo_c2_b) and not na(_lo_c3_b) and _lo_c3_b < _lo_c2_b
    bool a_high_sweep = not na(_hi_c2_a) and not na(_hi_c3_a) and _hi_c3_a > _hi_c2_a
    bool b_high_sweep = not na(_hi_c2_b) and not na(_hi_c3_b) and _hi_c3_b > _hi_c2_b
    bool bull_smt = (a_low_sweep and not b_low_sweep) or (b_low_sweep and not a_low_sweep)
    bool bear_smt = (a_high_sweep and not b_high_sweep) or (b_high_sweep and not a_high_sweep)
    [bull_smt, bear_smt]

// FUNCTION: BEARISH STATE MACHINE (SPEC-ONLY)

const int SM_IDLE = 0
const int SM_C3_TRACKING = 1
const int SM_CONFIRMED = 2
const int SM_INVALIDATED = 3
const int SM_EXPIRED = 4

f_tpd_state_bear(_htf, _show_tpd, _size) =>
    var int sm_state = SM_IDLE
    var bool bearish_valid_tpd = false
    var bool bearish_valid_tpd_ssmt = false
    var bool setup_confirmed = false
    var bool c2_high_swept = false
    var bool purge_seen = false
    var bool purge_is_main = false
    var bool failure_is_main = false
    var float htf_c2_high_main = na
    var float htf_c2_high_pair = na
    var float m15_tcisd_level = na
    var int m15_tcisd_ts = int(na)
    var int main_high_purge_ts = int(na)
    var int pair_high_purge_ts = int(na)
    var int c4_open_ts = int(na)
    var label bear_tpd_icon = na
    var label bear_c1_lbl = na
    var label bear_c2_lbl = na
    var label bear_c3_lbl = na
    var line bear_purge_line = na
    var line bear_tcisd_line = na
    var label bear_tcisd_lbl = na

    bool stage1_bear_evt = false
    bool stage2_bear_evt = false
    bool stage3_bear_evt = false
    bool invalidated_evt = false
    bool expired_evt = false
    bool dbg_crack1_out = false
    bool dbg_crack2_out = false
    bool dbg_con_hgh_out = false
    bool dbg_tie_rule_out = false
    bool dbg_purge_is_main_out = false
    bool dbg_failure_is_main_out = false
    bool dbg_c2_high_swept_out = false
    bool dbg_purge_seen_out = false
    bool dbg_in_c3_window_out = false

    if _show_tpd and timeframe.in_seconds(timeframe.period) <= timeframe.in_seconds(_htf)
        [mh2,ml2,mc2,mo2,mt2,mh1,ml1,mc1,mo1,mt1,mh0,ml0,mc0,mo0,mt0,mup1,mdw1,ml1t,mh1t] = f_main_c123(_htf)
        [oh2,ol2,oc2,oo2,ot2,oh1,ol1,oc1,oo1,ot1,oh0,ol0,oc0,oo0,ot0,oup1,odw1,ol1t,oh1t] = f_pair_c123(_htf, csym, xc0, xo0, xh0, xl0, xt0)

        bool htf_boundary = timeframe.change(_htf)
        int htf_span_sec = timeframe.in_seconds(_htf)
        int next_htf_open_ts = not na(htf_span_sec) ? mt0 + int(htf_span_sec * 1000) : int(na)
        bool is_m15_chart = timeframe.in_seconds(timeframe.period) == 15 * 60

        bool main_c2_up = mc1 > mo1
        bool main_c2_down = mc1 < mo1
        bool pair_c2_up = oc1 > oo1
        bool pair_c2_down = oc1 < oo1
        bool crack1 = (main_c2_up and pair_c2_down) or (main_c2_down and pair_c2_up)

        bool main_c3_open_above_c1_close = mo0 > mc2
        bool main_c3_open_below_c1_close = mo0 < mc2
        bool pair_c3_open_above_c1_close = oo0 > oc2
        bool pair_c3_open_below_c1_close = oo0 < oc2
        bool purge_is_main_now = main_c3_open_above_c1_close and pair_c3_open_below_c1_close
        bool failure_is_main_now = main_c3_open_below_c1_close and pair_c3_open_above_c1_close
        bool crack2 = (main_c3_open_above_c1_close and pair_c3_open_below_c1_close) or (main_c3_open_below_c1_close and pair_c3_open_above_c1_close)
        bool tie_rule = (mo0 == mc2) or (oo0 == oc2)

        hgh0 = array.from(mh2,mh1,mh0), hgh1 = array.from(oh2,oh1,oh0)
        bool con_hgh0 = array.max(hgh0) == mh1
        bool con_hgh1 = array.max(hgh1) == oh1
        bool con_hgh = failure_is_main_now ? con_hgh0 : purge_is_main_now ? con_hgh1 : false
        bool valid_bearish_tpd_now = crack1 and crack2 and con_hgh and not tie_rule
        dbg_crack1_out := crack1
        dbg_crack2_out := crack2
        dbg_con_hgh_out := con_hgh
        dbg_tie_rule_out := tie_rule

        bool main_sweeps_c2_high = not na(mh1) and high > mh1
        bool pair_sweeps_c2_high = not na(oh1) and xh0 > oh1
        bool either_sweeps_c2_high = main_sweeps_c2_high or pair_sweeps_c2_high
        int bear_potential_tcisd_ts = failure_is_main ? pair_high_purge_ts : main_high_purge_ts
        int bear_potential_tcisd_back = int(ta.barssince(time == bear_potential_tcisd_ts))

        if htf_boundary
            // C4 open reset behavior for prior active instance.
            if sm_state == SM_C3_TRACKING and not setup_confirmed
                bearish_valid_tpd := false
                bearish_valid_tpd_ssmt := false
                setup_confirmed := false
                c2_high_swept := false
                purge_seen := false
                purge_is_main := false
                failure_is_main := false
                htf_c2_high_main := na
                htf_c2_high_pair := na
                m15_tcisd_level := na
                m15_tcisd_ts := int(na)
                main_high_purge_ts := int(na)
                pair_high_purge_ts := int(na)
                c4_open_ts := int(na)
                if not na(bear_tpd_icon)
                    label.delete(bear_tpd_icon)
                    bear_tpd_icon := na
                if not na(bear_c1_lbl)
                    label.delete(bear_c1_lbl)
                    bear_c1_lbl := na
                if not na(bear_c2_lbl)
                    label.delete(bear_c2_lbl)
                    bear_c2_lbl := na
                if not na(bear_c3_lbl)
                    label.delete(bear_c3_lbl)
                    bear_c3_lbl := na
                if not na(bear_purge_line)
                    line.delete(bear_purge_line)
                    bear_purge_line := na
                if not na(bear_tcisd_line)
                    line.delete(bear_tcisd_line)
                    bear_tcisd_line := na
                if not na(bear_tcisd_lbl)
                    label.delete(bear_tcisd_lbl)
                    bear_tcisd_lbl := na
                expired_evt := true
                sm_state := SM_EXPIRED
            else if sm_state == SM_CONFIRMED
                // For confirmed setups, reset runtime state at C4 open but keep drawings.
                bearish_valid_tpd := false
                bearish_valid_tpd_ssmt := false
                setup_confirmed := false
                c2_high_swept := false
                purge_seen := false
                purge_is_main := false
                failure_is_main := false
                htf_c2_high_main := na
                htf_c2_high_pair := na
                m15_tcisd_level := na
                m15_tcisd_ts := int(na)
                main_high_purge_ts := int(na)
                pair_high_purge_ts := int(na)
                c4_open_ts := int(na)
                sm_state := SM_IDLE

            if sm_state == SM_EXPIRED or sm_state == SM_INVALIDATED
                sm_state := SM_IDLE

            // IDLE -> C3_TRACKING entry gate.
            if sm_state == SM_IDLE and valid_bearish_tpd_now
                purge_is_main := purge_is_main_now
                failure_is_main := failure_is_main_now
                if purge_is_main or failure_is_main
                    bearish_valid_tpd := true
                    bearish_valid_tpd_ssmt := false
                    setup_confirmed := false
                    c2_high_swept := false
                    purge_seen := false
                    htf_c2_high_main := mh1
                    htf_c2_high_pair := oh1
                    m15_tcisd_level := na
                    m15_tcisd_ts := int(na)
                    main_high_purge_ts := int(na)
                    pair_high_purge_ts := int(na)
                    c4_open_ts := next_htf_open_ts
                    if not na(bear_tpd_icon)
                        label.delete(bear_tpd_icon)
                    if not na(bear_c1_lbl)
                        label.delete(bear_c1_lbl)
                    if not na(bear_c2_lbl)
                        label.delete(bear_c2_lbl)
                    if not na(bear_c3_lbl)
                        label.delete(bear_c3_lbl)
                    bear_tpd_icon := label.new(mh1t, mh1, text=f_text(_htf) + "\n" + tpd_bear_shape, xloc=xloc.bar_time, yloc=yloc.abovebar, color=color.new(color.gray, 100), size=_size, textcolor=tpd_bear_color, tooltip="Bearish Valid TPD", text_font_family=target_font)
                    array.push(tpd_labels, bear_tpd_icon)
                    if array.size(tpd_labels) > max_tpd_bars
                        label.delete(array.shift(tpd_labels))
                    bear_c1_lbl := label.new(mt2, mh2, "C1", xloc=xloc.bar_time, yloc=yloc.abovebar, style=label.style_label_down, size=size.tiny, color=color.new(color.white, 100), textcolor=tpd_bear_color, tooltip="Bearish TPD C1", text_font_family=target_font)
                    array.push(tpd_labels, bear_c1_lbl)
                    if array.size(tpd_labels) > max_tpd_bars
                        label.delete(array.shift(tpd_labels))
                    bear_c2_lbl := label.new(mt1, mh1, "C2", xloc=xloc.bar_time, yloc=yloc.abovebar, style=label.style_label_down, size=size.tiny, color=color.new(color.white, 100), textcolor=tpd_bear_color, tooltip="Bearish TPD C2", text_font_family=target_font)
                    array.push(tpd_labels, bear_c2_lbl)
                    if array.size(tpd_labels) > max_tpd_bars
                        label.delete(array.shift(tpd_labels))
                    bear_c3_lbl := label.new(mt0, mh0, "C3", xloc=xloc.bar_time, yloc=yloc.abovebar, style=label.style_label_down, size=size.tiny, color=color.new(color.white, 100), textcolor=tpd_bear_color, tooltip="Bearish TPD C3", text_font_family=target_font)
                    array.push(tpd_labels, bear_c3_lbl)
                    if array.size(tpd_labels) > max_tpd_bars
                        label.delete(array.shift(tpd_labels))
                    stage1_bear_evt := true
                    sm_state := SM_C3_TRACKING

        if sm_state == SM_C3_TRACKING
            bool failure_sweeps_c2_high = failure_is_main ? main_sweeps_c2_high : pair_sweeps_c2_high
            bool purge_sweeps_c2_high = purge_is_main ? main_sweeps_c2_high : pair_sweeps_c2_high
            bool is_h6_chart = timeframe.in_seconds(timeframe.period) == 6 * 60 * 60

            if either_sweeps_c2_high
                c2_high_swept := true
            if na(main_high_purge_ts) and not na(mh1) and high > mh1
                main_high_purge_ts := time
            if na(pair_high_purge_ts) and not na(oh1) and xh0 > oh1
                pair_high_purge_ts := time

            // Invalidation must win over confirmation.
            if failure_sweeps_c2_high and not setup_confirmed
                bearish_valid_tpd := false
                bearish_valid_tpd_ssmt := false
                setup_confirmed := false
                c2_high_swept := false
                purge_seen := false
                purge_is_main := false
                failure_is_main := false
                htf_c2_high_main := na
                htf_c2_high_pair := na
                m15_tcisd_level := na
                m15_tcisd_ts := int(na)
                main_high_purge_ts := int(na)
                pair_high_purge_ts := int(na)
                c4_open_ts := int(na)
                if not na(bear_tpd_icon)
                    label.delete(bear_tpd_icon)
                    bear_tpd_icon := na
                if not na(bear_c1_lbl)
                    label.delete(bear_c1_lbl)
                    bear_c1_lbl := na
                if not na(bear_c2_lbl)
                    label.delete(bear_c2_lbl)
                    bear_c2_lbl := na
                if not na(bear_c3_lbl)
                    label.delete(bear_c3_lbl)
                    bear_c3_lbl := na
                if not na(bear_purge_line)
                    line.delete(bear_purge_line)
                    bear_purge_line := na
                if not na(bear_tcisd_line)
                    line.delete(bear_tcisd_line)
                    bear_tcisd_line := na
                if not na(bear_tcisd_lbl)
                    label.delete(bear_tcisd_lbl)
                    bear_tcisd_lbl := na
                invalidated_evt := true
                sm_state := SM_INVALIDATED

            if sm_state == SM_C3_TRACKING and c2_high_swept and purge_sweeps_c2_high and not purge_seen
                purge_seen := true
                bearish_valid_tpd_ssmt := true
                stage2_bear_evt := true
                float purge_level = purge_is_main ? htf_c2_high_main : htf_c2_high_pair
                int purge_time = purge_is_main ? mh1t : oh1t
                int purge_end_time = na(c4_open_ts) ? time : c4_open_ts
                if not na(bear_purge_line)
                    line.delete(bear_purge_line)
                bear_purge_line := line.new(purge_time, purge_level, purge_end_time, purge_level, xloc=xloc.bar_time, extend=extend.none, color=color.new(tpd_bear_color, 50), width=1)
                array.push(purge_lines, bear_purge_line)
                if array.size(purge_lines) > max_purge_lines
                    line.delete(array.shift(purge_lines))

            if sm_state == SM_C3_TRACKING and purge_seen and na(m15_tcisd_level) and not na(bear_potential_tcisd_back)
                int bear_fail_back0 = bear_potential_tcisd_back
                int bear_fail_back1 = bear_potential_tcisd_back + 1
                if failure_is_main
                    bool bear_fail_match0 = bear_fail_back0 <= bar_index and close[bear_fail_back0] > open[bear_fail_back0]
                    bool bear_fail_match1 = bear_fail_back1 <= bar_index and close[bear_fail_back1] > open[bear_fail_back1]
                    if bear_fail_match0
                        m15_tcisd_level := open[bear_fail_back0]
                        m15_tcisd_ts := int(time[bear_fail_back0])
                    else if bear_fail_match1
                        m15_tcisd_level := open[bear_fail_back1]
                        m15_tcisd_ts := int(time[bear_fail_back1])
                else
                    bool bear_fail_match0 = bear_fail_back0 <= bar_index and xc0[bear_fail_back0] > xo0[bear_fail_back0]
                    bool bear_fail_match1 = bear_fail_back1 <= bar_index and xc0[bear_fail_back1] > xo0[bear_fail_back1]
                    if bear_fail_match0
                        m15_tcisd_level := xo0[bear_fail_back0]
                        m15_tcisd_ts := int(time[bear_fail_back0])
                    else if bear_fail_match1
                        m15_tcisd_level := xo0[bear_fail_back1]
                        m15_tcisd_ts := int(time[bear_fail_back1])

            if sm_state == SM_C3_TRACKING and purge_seen and is_m15_chart and not na(m15_tcisd_level) and not na(m15_tcisd_ts) and not na(c4_open_ts) and na(bear_tcisd_line)
                if array.size(purge_lines) >= max_purge_lines
                    line.delete(array.shift(purge_lines))
                bear_tcisd_line := line.new(m15_tcisd_ts, m15_tcisd_level, c4_open_ts, m15_tcisd_level, xloc=xloc.bar_time, extend=extend.none, color=tcisd_level_color, width=1)
                array.push(purge_lines, bear_tcisd_line)
                if not na(bear_tcisd_lbl)
                    label.delete(bear_tcisd_lbl)
                bear_tcisd_lbl := label.new(c4_open_ts, m15_tcisd_level, "m15 tCISD", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_label_left, size=size.tiny, color=color.new(color.white, 100), textcolor=color.black, text_font_family=target_font)
                array.push(tpd_labels, bear_tcisd_lbl)
                if array.size(tpd_labels) > max_tpd_bars
                    label.delete(array.shift(tpd_labels))

            bool in_c3_window = na(c4_open_ts) ? true : time < c4_open_ts
            if sm_state == SM_C3_TRACKING and purge_seen and in_c3_window and not na(m15_tcisd_level)
                bool confirm_now = false
                if is_m15_chart
                    float bear_fail_c_now = failure_is_main ? close : xc0
                    confirm_now := barstate.isconfirmed and bear_fail_c_now < m15_tcisd_level
                else if is_h6_chart
                    float main_m15_close_min = na
                    float pair_m15_close_min = na
                    array<float> main_m15_close_arr = request.security_lower_tf(sym, "15", close)
                    array<float> pair_m15_close_arr = request.security_lower_tf(csym, "15", close, ignore_invalid_symbol = true)
                    if array.size(main_m15_close_arr) > 0
                        main_m15_close_min := array.min(main_m15_close_arr)
                    if array.size(pair_m15_close_arr) > 0
                        pair_m15_close_min := array.min(pair_m15_close_arr)
                    float failure_m15_close_min = failure_is_main ? main_m15_close_min : pair_m15_close_min
                    confirm_now := not na(failure_m15_close_min) and failure_m15_close_min < m15_tcisd_level
                if confirm_now
                    setup_confirmed := true
                    stage3_bear_evt := true
                    sm_state := SM_CONFIRMED

        if sm_state == SM_INVALIDATED
            sm_state := SM_IDLE
        if sm_state == SM_EXPIRED
            sm_state := SM_IDLE
        dbg_purge_is_main_out := purge_is_main
        dbg_failure_is_main_out := failure_is_main
        dbg_c2_high_swept_out := c2_high_swept
        dbg_purge_seen_out := purge_seen
        dbg_in_c3_window_out := na(c4_open_ts) ? true : time < c4_open_ts

    [bearish_valid_tpd, bearish_valid_tpd_ssmt, setup_confirmed, stage1_bear_evt, stage2_bear_evt, stage3_bear_evt, invalidated_evt, expired_evt, sm_state, dbg_crack1_out, dbg_crack2_out, dbg_con_hgh_out, dbg_tie_rule_out, dbg_purge_is_main_out, dbg_failure_is_main_out, dbg_c2_high_swept_out, dbg_purge_seen_out, dbg_in_c3_window_out]

// ---------------------------------------------------------------
// SPEC-ONLY EXECUTION (H6->m15, BEARISH)
// ---------------------------------------------------------------
is_m15_model_chart = timeframe.in_seconds(timeframe.period) == 15 * 60
is_h6_model_chart = timeframe.in_seconds(timeframe.period) == 6 * 60 * 60
model_chart_ok = is_m15_model_chart or is_h6_model_chart
tpd_active = dyad_selection_mode != "Off" and model_chart_ok

[bearishValidTPD, bearishValidTPD_SSMT, setupConfirmed, stage1_bear, stage2_bear, stage3_bear, invalidated_now, expired_now, sm_state_now, dbg_crack1, dbg_crack2, dbg_con_hgh, dbg_tie_rule, dbg_purge_is_main, dbg_failure_is_main, dbg_c2_high_swept, dbg_purge_seen, dbg_in_c3_window] = f_tpd_state_bear(TPD_SCAN_TF, tpd_active, tpd_lbl_size)

scan_tf_alert = f_text(TPD_SCAN_TF)
alert_time = f_fmt_time(time)

alertcondition(stage1_bear, "TPDSSMT Stage 1 Bearish TPD", "Stage 1: HTF C3 open with bearish valid TPD")
alertcondition(stage2_bear, "TPDSSMT Stage 2 Bearish Valid TPD/SSMT", "Stage 2: Bearish purge complete (Valid TPD/SSMT)")
alertcondition(stage3_bear, "TPDSSMT Stage 3 Bearish Setup Confirmed", "Stage 3: Bearish setup confirmed (m15 tCISD)")

if stage1_bear
    alert(syminfo.ticker + ", " + scan_tf_alert + ", Bearish Valid TPD, " + alert_time, alert.freq_once_per_bar_close)
if stage2_bear
    alert(syminfo.ticker + ", " + scan_tf_alert + ", Bearish Valid TPD/SSMT, " + alert_time, alert.freq_once_per_bar_close)
if stage3_bear
    alert(syminfo.ticker + ", " + scan_tf_alert + ", Bearish Setup Confirmed (tCISD), " + alert_time, alert.freq_once_per_bar_close)

        table.cell(sm_state_tbl, 3, 2, bearishValidTPD ? "Reached" : "--", text_color=color.black, bgcolor=level1_bg, text_size=size.small)

        table.cell(sm_state_tbl, 0, 3, "Time Phase", text_color=color.black, bgcolor=body_bg, text_size=size.small)
