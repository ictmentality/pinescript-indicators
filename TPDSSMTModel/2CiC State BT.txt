f_check_match(input_sym) =>
    if str.length(input_sym) == 0 or not str.contains(input_sym, ":")
        false
    else
        t_sym = array.get(str.split(input_sym, ":"), 1)
        (current_ticker == t_sym) or str.contains(sym, t_sym) or str.contains(input_sym, current_ticker)

f_text(string _tf) =>
    float _tf_num = str.tonumber(_tf)
    int _tf_int = int(na)
    string _text = _tf
    if not na(_tf_num)
        _tf_int := int(_tf_num)
    if _tf == "1D"
        _text := "D"
    else if _tf == "1W"
        _text := "W"
    else if _tf == "1M"
        _text := "M"
    else if not na(_tf_int)
        if _tf_int < 60
            _text := "m" + _tf
        else if _tf_int % 60 == 0
            _text := "H" + str.tostring(_tf_int / 60)
        else
            _text := "m" + _tf
    _text

f_yn(bool _v) =>
    _v ? "Y" : "N"

f_sm_state_text(_s) =>
    _s == 0 ? "IDLE" : _s == 1 ? "C3_TRACKING" : "UNKNOWN"

f_snap_to_containing_bar_open(int _ts) =>
    int snapped = ta.valuewhen(time <= _ts and time_close > _ts, time, 0)
    na(snapped) ? _ts : snapped

f_qt_pick_qr(int idx, qt.QuarterRecord q1, qt.QuarterRecord q2, qt.QuarterRecord q3, qt.QuarterRecord q4) =>
    qt.QuarterRecord out = q1
    if idx == 2
        out := q2
    else if idx == 3
        out := q3
    else if idx == 4
        out := q4
    out

f_qt_daily_c123() =>
    var qt.QTConfig cfg = qt.qt_config_default()
    var qt.CycleState st = qt.qt_state_new()
    [st0, d] = qt.qt_daily_update(st, cfg, time, time_close)
    st := st0

    qt.QuarterRecord qcur = f_qt_pick_qr(d.curQuarterIndex, d.q1, d.q2, d.q3, d.q4)

        bool pair_c3_open_below_c1_close = oo0 < oc2
        bool crack2 = have_hist and ((main_c3_open_above_c1_close and pair_c3_open_below_c1_close) or (main_c3_open_below_c1_close and pair_c3_open_above_c1_close))

        float bear_fail_c2_o_now = failure_is_main_now ? mo1 : oo1
        float bear_fail_c2_c_now = failure_is_main_now ? mc1 : oc1

        float bear_c2_upper_wick_now = bear_fail_c2_h_now - math.max(bear_fail_c2_o_now, bear_fail_c2_c_now)

        dbg_crack2_out := crack2

            if sm_state == SM_C3_TRACKING and not setup_confirmed

                bearish_valid_tpd_ssmt := false
                setup_confirmed := false

                failure_is_main := false

                htf_c2_high_pair := na
                m15_tcisd_level := na
                m15_tcisd_ts := int(na)
                m15_tcisd_purge_level := na
                m15_tcisd_purge_ts := int(na)
                purge_line_level := na
                purge_line_ts := int(na)
                setup_inv_level := na
                setup_inv_ts := int(na)
                expansion_inv_level := na
                main_high_purge_ts := int(na)

                failure_is_main := false

                purge_seen := false

                htf_c2_high_pair := na
                m15_tcisd_level := na
                m15_tcisd_ts := int(na)
                m15_tcisd_purge_level := na
                m15_tcisd_purge_ts := int(na)
                purge_line_level := na
                purge_line_ts := int(na)
                setup_inv_level := na
                setup_inv_ts := int(na)
                expansion_inv_level := na
                main_high_purge_ts := int(na)
                pair_high_purge_ts := int(na)
                c4_open_ts := int(na)

                bearish_valid_tpd_ssmt := false
                setup_confirmed := false
                purge_is_main := purge_is_main_now
                failure_is_main := failure_is_main_now
                c2_high_swept := false
                purge_seen := false
                htf_c2_high_main := mh1
                htf_c2_high_pair := oh1
                m15_tcisd_level := na
                m15_tcisd_ts := int(na)
                m15_tcisd_purge_level := na
                m15_tcisd_purge_ts := int(na)
                purge_line_level := purge_is_main_now ? mh1 : oh1
                purge_line_ts := purge_is_main_now ? mh1t : oh1t
                setup_inv_level := failure_is_main_now ? mh1 : oh1
                setup_inv_ts := failure_is_main_now ? mt0 : ot0
                expansion_inv_level := bear_c2_expansion_invalidation_now
                main_high_purge_ts := int(na)
                pair_high_purge_ts := int(na)
                c4_open_ts := next_htf_open_ts
                stage1_seq += 1
                stage1_ts := mh1t
                stage1_price := mh1
                sm_state := SM_C3_TRACKING

        bool main_sweeps_c2_high = not na(htf_c2_high_main) and not na(mh0) and mh0 > htf_c2_high_main
        bool pair_sweeps_c2_high = not na(htf_c2_high_pair) and not na(oh0) and oh0 > htf_c2_high_pair

        int bear_potential_tcisd_back = na(bear_potential_tcisd_match_bar) ? int(na) : bar_index - bear_potential_tcisd_match_bar
        int bear_potential_purge_tcisd_ts = purge_is_main ? main_high_purge_ts : pair_high_purge_ts

        bool bear_purge_tcisd_back_valid = not na(bear_potential_purge_tcisd_back) and bear_potential_purge_tcisd_back >= 0 and bear_potential_purge_tcisd_back <= TCISD_MAX_LOOKBACK_BARS

        if sm_state == SM_C3_TRACKING
            bool either_sweeps_c2_high = main_sweeps_c2_high or pair_sweeps_c2_high
            bool failure_sweeps_c2_high = failure_is_main ? main_sweeps_c2_high : pair_sweeps_c2_high
