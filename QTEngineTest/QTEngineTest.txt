// -----------------------------------------------------------------------------
// Helpers (display-only)
// -----------------------------------------------------------------------------
f_hhmmss(int ts, string tz) =>
    int hh = hour(ts, tz)
    int mm = minute(ts, tz)
    int ss = second(ts, tz)
    float(hh * 10000 + mm * 100 + ss)

f_fmt(float v) =>
    na(v) ? "-" : str.tostring(v)

f_fmt_time(int ts, string tz) =>
    na(ts) ? "-" : str.tostring(f_hhmmss(ts, tz))

f_fmt_ts(int ts, string tz) =>
    na(ts) ? "-" : str.format_time(ts, "yyyy-MM-dd HH:mm", tz)

f_ceil_5s(int ts) =>
    int step = 5 * 1000
    int rem = ts % step
    rem == 0 ? ts : ts + (step - rem)

// Snap an arbitrary timestamp to the current chart bar open (for visual anchoring on higher TFs).
f_snap_to_chart_bar(int ts) =>
    int snap = ta.valuewhen(time <= ts and time_close > ts, time, 0)
    na(ts) ? ts : (na(snap) ? ts : snap)

// QTIndicator mapping (string → TradingView style)
lineStyleFrom(string s) =>
    switch s
        "Solid"  => line.style_solid
        "Dashed" => line.style_dashed
        =>          line.style_dotted

f_clamp_int(int v, int lo, int hi) =>
    v < lo ? lo : v > hi ? hi : v

f_pick_qr(int idx, qt.QuarterRecord q1, qt.QuarterRecord q2, qt.QuarterRecord q3, qt.QuarterRecord q4) =>
    qt.QuarterRecord out = q1
    if idx == 2
        out := q2
    else if idx == 3
        out := q3
    else if idx == 4
        out := q4
    out

f_qr_ready(qt.QuarterRecord q) =>
    q.has and not q.isEmpty and not na(q.h) and not na(q.hTs) and not na(q.boundaryTs) and q.hTs >= q.boundaryTs

f_qr_ready_low(qt.QuarterRecord q) =>
    q.has and not q.isEmpty and not na(q.l) and not na(q.lTs) and not na(q.boundaryTs) and q.lTs >= q.boundaryTs

f_qr_ready_bh(qt.QuarterRecord q) =>
    q.has and not q.isEmpty and not na(q.bh) and not na(q.bhTs) and not na(q.boundaryTs) and q.bhTs >= q.boundaryTs

f_qr_ready_bl(qt.QuarterRecord q) =>
    q.has and not q.isEmpty and not na(q.bl) and not na(q.blTs) and not na(q.boundaryTs) and q.blTs >= q.boundaryTs

f_ssmt_daily_highs() =>
    var qt.QTConfig cfg = qt.qt_config_default()
    var qt.CycleState st = qt.qt_state_new()
    int t = time
    int tc = time_close
    [st0, d] = qt.qt_daily_update(st, cfg, t, tc)
    st := st0

    qt.QuarterRecord qcur = f_pick_qr(d.curQuarterIndex, d.q1, d.q2, d.q3, d.q4)
    qt.QuarterRecord qprev = d.curQuarterIndex == 1 ? st.daily.prev1 : d.curQuarterIndex == 2 ? d.q1 : d.curQuarterIndex == 3 ? d.q2 : d.q3
    float hcur = f_qr_ready(qcur) ? qcur.h : na
    int   hcurTs = f_qr_ready(qcur) ? qcur.hTs : na
    float hprev = f_qr_ready(qprev) ? qprev.h : na
    int   hprevTs = f_qr_ready(qprev) ? qprev.hTs : na
    float bhcur = f_qr_ready_bh(qcur) ? qcur.bh : na
    int   bhcurTs = f_qr_ready_bh(qcur) ? qcur.bhTs : na
    float bhprev = f_qr_ready_bh(qprev) ? qprev.bh : na
    int   bhprevTs = f_qr_ready_bh(qprev) ? qprev.bhTs : na

    int qcurTs = d.q1Ts
    if d.curQuarterIndex == 2
        qcurTs := d.q2Ts
    else if d.curQuarterIndex == 3
        qcurTs := d.q3Ts
    else if d.curQuarterIndex == 4
        qcurTs := d.q4Ts

    [hcur, hcurTs, hprev, hprevTs, bhcur, bhcurTs, bhprev, bhprevTs, d.curQuarterIndex, qcurTs]

f_ssmt_sym_ticker(string sym) =>
    string s = str.upper(sym)
    string[] parts = str.split(s, ":")
    array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : s

    string s = f_ssmt_sym_ticker(sym)
    string tag = s
    if str.contains(s, "MES")
        tag := "ES"
    else if str.contains(s, "MNQ")
        tag := "NQ"
    else if str.contains(s, "MYM")
        tag := "YM"
    else if str.contains(s, "ES")
        tag := "ES"
    else if str.contains(s, "NQ")
        tag := "NQ"
    else if str.contains(s, "YM")
        tag := "YM"
    tag

f_ssmt_hist_push(line ln, label lb, line[] lines, label[] labels, int maxN) =>
    if not na(ln) or not na(lb)
        if maxN <= 0
            if not na(ln)
                line.delete(ln)
            if not na(lb)
                label.delete(lb)
        else
            array.push(lines, ln)
            array.push(labels, lb)
            while array.size(lines) > maxN
                line oldLn = array.shift(lines)
                if not na(oldLn)
                    line.delete(oldLn)
                label oldLb = array.shift(labels)
                if not na(oldLb)
                    label.delete(oldLb)
    0

f_ssmt_hist_trim(line[] lines, label[] labels, int maxN) =>
    while array.size(lines) > maxN
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    while array.size(labels) > maxN
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

f_ssmt_hist_clear(line[] lines, label[] labels) =>
    while array.size(lines) > 0
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
    while array.size(labels) > 0
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

// Array helpers

f_find_int(int[] arr, int v) =>
    int idx = -1
    int i = array.size(arr) - 1
    while i >= 0 and idx == -1
        if array.get(arr, i) == v
            idx := i
        i -= 1
    idx

f_push_unique_int(int[] arr, int v, int maxN) =>
    if f_find_int(arr, v) == -1
        array.push(arr, v)
        while array.size(arr) > maxN
            array.shift(arr)
    0

// -----------------------------------------------------------------------------
// Old-QT vertical divider line storage helpers
// -----------------------------------------------------------------------------
type vset
    line[] ln
    int[]  ts

vset_new() =>
    vset.new(array.new_line(), array.new_int())

vset_clear(vset s) =>
    int i = array.size(s.ln) - 1
    while i >= 0
        line.delete(array.get(s.ln, i))
        i -= 1
    array.clear(s.ln)
    array.clear(s.ts)
    0

vset_has(vset s, int ts) =>
    bool found = false
    int i = array.size(s.ts) - 1
    while i >= 0 and not found
        found := array.get(s.ts, i) == ts
        i -= 1
    found

vset_add(vset s, int ts, color c) =>
    if not vset_has(s, ts)
        array.push(s.ln, line.new(ts, low, ts, high, xloc=xloc.bar_time, extend=extend.both, style=line.style_solid, width=1, color=f_vcol(c)))
        array.push(s.ts, ts)
    0

vset_trim_since(vset s, int cutoffTs) =>
    int i = array.size(s.ts) - 1
    while i >= 0
        int t = array.get(s.ts, i)
        if t < cutoffTs
            line ln = array.remove(s.ln, i)
            line.delete(ln)
            array.remove(s.ts, i)
        i -= 1
    0

f_set_or_make_vline(line ln, int ts, color c) =>
    line out = ln
    if na(out)
        out := line.new(ts, low, ts, high, xloc=xloc.bar_time, extend=extend.both, style=line.style_solid, width=1, color=f_vcol(c))
    else
        line.set_xy1(out, ts, low)
        line.set_xy2(out, ts, high)
        line.set_style(out, line.style_solid)
        line.set_width(out, 1)
        line.set_color(out, f_vcol(c))
    out

            else if isChartNQNHB
                curL := nqBLcurN
                curLTs := nqBLcurTsN
                prevL := nqBLprevN
                prevLTs := nqBLprevTsN
                curQTs := nqQtsN
                curSweep := nqSweepNHB
            int prevLTsSnap = isChartESNHB ? esBLprevTsNSnap : isChartNQNHB ? nqBLprevTsNSnap : prevLTs

            bool hasPointsB = splitNHB and not na(prevL) and not na(prevLTs) and not na(curQTs)
            if splitNHB
                int eventKeyB = prevLTs
                if na(nanoHssmtBEventKey) or nanoHssmtBEventKey != eventKeyB
                    f_ssmt_hist_push(nanoHssmtBLine, nanoHssmtBLbl, nanoHssmtBHistLines, nanoHssmtBHistLbls, ssmtKeepEff)
                    nanoHssmtBLine := na
                    nanoHssmtBLbl := na
                    nanoHssmtBEventKey := eventKeyB
                    nanoHssmtBActiveQTs := curQTs
                    nanoHssmtBExtendTier := 0

                if hasPointsB and curSweep
                    int thirdLenB = int(NanoSsmtQLen / 3)
                    int q1EndB = curQTs + thirdLenB
                    int q2EndB = curQTs + (thirdLenB * 2)
                    int q3EndB = curQTs + NanoSsmtQLen

                    int sweepTierB = 1
                    if not na(curLTs)
                        if curLTs >= q2EndB
                            sweepTierB := 3
                        else if curLTs >= q1EndB
                            sweepTierB := 2

                    int barTierB = 1
                    if time_close >= q2EndB
                        barTierB := 3
                    else if time_close >= q1EndB
                        barTierB := 2

                    nanoHssmtBExtendTier := nanoHssmtBExtendTier < sweepTierB ? sweepTierB : nanoHssmtBExtendTier
                    bool touchB = low <= prevL
                    if touchB
                        nanoHssmtBExtendTier := nanoHssmtBExtendTier < barTierB ? barTierB : nanoHssmtBExtendTier

                    int x2B = q1EndB
                    if nanoHssmtBExtendTier >= 3
                        x2B := q3EndB
                    else if nanoHssmtBExtendTier == 2
                        x2B := q2EndB

                    nanoHssmtBLine := f_set_or_make_hline_w(nanoHssmtBLine, prevLTsSnap, x2B, prevL, color.new(ssmtBullLineCol, 50), line.style_solid, ssmtLineWidth)
                else if hasPointsB and not curSweep and not na(curL) and not na(curLTs)
                    nanoHssmtBLine := f_set_or_make_line(nanoHssmtBLine, prevLTs, prevL, curLTs, curL, color.new(ssmtBullLineCol, 50), ssmtLineWidth)
                bool showLabelB = (isChartESNHB and crackedESNHB) or (isChartNQNHB and crackedNQNHB)
                if hasPointsB and crackedNHB != "" and showLabelB
                    string lblTextB = "(H) Nano " + crackedNHB

                    int xMidB = prevLTs
                    float yMidB = prevL
                    if not na(nanoHssmtBLine)
                        int x1B = line.get_x1(nanoHssmtBLine)
                        int x2B = line.get_x2(nanoHssmtBLine)
                        xMidB := x1B + int((x2B - x1B) / 2)
                        yMidB := (line.get_y1(nanoHssmtBLine) + line.get_y2(nanoHssmtBLine)) / 2
                    float yLblB = yMidB - syminfo.mintick

                    if na(nanoHssmtBLbl)
                        nanoHssmtBLbl := label.new(xMidB, yLblB, lblTextB, xloc=xloc.bar_time, style=label.style_none, textcolor=color.black, size=size.tiny)
                    else
                        label.set_xy(nanoHssmtBLbl, xMidB, yLblB)
                        label.set_text(nanoHssmtBLbl, lblTextB)
                        label.set_textcolor(nanoHssmtBLbl, color.black)
                else if not showLabelB and not na(nanoHssmtBLbl)
                    label.delete(nanoHssmtBLbl)
                    nanoHssmtBLbl := na
            if not splitNHB and not na(nanoHssmtBEventKey)
                bool rolledB = not na(nanoHssmtBActiveQTs) and not na(curQTs) and curQTs != nanoHssmtBActiveQTs
                if rolledB
                    f_ssmt_hist_push(nanoHssmtBLine, nanoHssmtBLbl, nanoHssmtBHistLines, nanoHssmtBHistLbls, ssmtKeepEff)
                else
                    if not na(nanoHssmtBLine)
                        line.delete(nanoHssmtBLine)
                    if not na(nanoHssmtBLbl)
                        label.delete(nanoHssmtBLbl)
                nanoHssmtBLine := na
                nanoHssmtBLbl := na
                nanoHssmtBEventKey := na
                nanoHssmtBActiveQTs := na
                nanoHssmtBExtendTier := 0
        if not (hssmtReadyNHB and isPairChartNHB)
            // Not ready or not on ES/NQ: clear active state to avoid stale drawings
            if not na(nanoHssmtBLine)
                line.delete(nanoHssmtBLine)
                nanoHssmtBLine := na
            if not na(nanoHssmtBLbl)
                label.delete(nanoHssmtBLbl)
                nanoHssmtBLbl := na
            nanoHssmtBEventKey := na
            nanoHssmtBExtendTier := 0
        f_ssmt_hist_trim(nanoHssmtBHistLines, nanoHssmtBHistLbls, ssmtKeepEff)
    else
        if not na(nanoHssmtBLine)
            line.delete(nanoHssmtBLine)
            nanoHssmtBLine := na
        if not na(nanoHssmtBLbl)
            label.delete(nanoHssmtBLbl)
            nanoHssmtBLbl := na
        f_ssmt_hist_clear(nanoHssmtBHistLines, nanoHssmtBHistLbls)
        nanoHssmtBEventKey := na
        nanoHssmtBExtendTier := 0




    0
