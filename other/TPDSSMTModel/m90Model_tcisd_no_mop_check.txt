f_ticker_only(string s) =>
    str.contains(s, ":") ? array.get(str.split(s, ":"), 1) : s

f_sym_ticker(string input_sym) =>
    string s = str.upper(input_sym)
    if str.contains(s, "\"SYMBOL\":\"")
        string[] partsSym = str.split(s, "\"SYMBOL\":\"")
        if array.size(partsSym) > 1
            string right = array.get(partsSym, 1)
            string[] partsEnd = str.split(right, "\"")
            s := array.get(partsEnd, 0)
    if str.contains(s, "{")
        string[] partsBrace = str.split(s, "{")
        s := array.get(partsBrace, 0)
    if str.contains(s, "|")
        string[] partsPipe = str.split(s, "|")
        s := array.get(partsPipe, 0)
    string[] parts = str.split(s, ":")
    array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : s

f_match_tag(string input_sym) =>
    string tk = str.upper(f_sym_ticker(input_sym))
    string tag = ""
    if str.contains(tk, "DXY")
        tag := "DXY"
    else if str.contains(tk, "MNQ") or str.contains(tk, "NQ")
        tag := "NQ"
    else if str.contains(tk, "MES") or str.contains(tk, "ES")
        tag := "ES"
    else if str.contains(tk, "MYM") or str.contains(tk, "YM")
        tag := "YM"
    else if str.contains(tk, "MCL") or str.contains(tk, "CL")
        tag := "CL"
    else if str.contains(tk, "MRB") or str.contains(tk, "RB")
        tag := "RB"
    else if str.contains(tk, "M6E") or str.contains(tk, "6E")
        tag := "6E"
    else if str.contains(tk, "M6B") or str.contains(tk, "6B")
        tag := "6B"
    else if str.contains(tk, "MGC") or str.contains(tk, "GC")
        tag := "GC"
    else if str.contains(tk, "SIL") or str.contains(tk, "SI")
        tag := "SI"
    tag

f_check_match(string input_sym) =>
    bool valid = str.length(input_sym) > 0
    if not valid
        false
    else
        string inputUp = str.upper(input_sym)
        string tSym = f_sym_ticker(input_sym)
        string curTicker = str.upper(syminfo.ticker)
        string curId = str.upper(syminfo.tickerid)
        bool directMatch = (curTicker == tSym) or str.contains(curId, tSym) or str.contains(inputUp, curTicker)
        if directMatch
            true
        else
            string inTag = f_match_tag(input_sym)
            string curTag = f_match_tag(syminfo.tickerid)
            str.length(inTag) > 0 and str.length(curTag) > 0 and inTag == curTag

f_pair(string a, string b) =>
    f_check_match(a) ? b : f_check_match(b) ? a : ""

f_pick_qr(int idx, qt.QuarterRecord q1, qt.QuarterRecord q2, qt.QuarterRecord q3, qt.QuarterRecord q4) =>
    qt.QuarterRecord out = q1
    if idx == 2
        out := q2
    else if idx == 3
        out := q3
    else if idx == 4
        out := q4
    out

f_sixh_session_end_ts(int src_ts) =>
    string tz = syminfo.timezone
    int y = year(src_ts, tz)
    int m = month(src_ts, tz)
    int d = dayofmonth(src_ts, tz)
    int h = hour(src_ts, tz)
    int day_start = timestamp(tz, y, m, d, 0, 0)
    int sixh_ms = 6 * 60 * 60 * 1000
    int out = h < 6 ? day_start + sixh_ms : h < 12 ? day_start + sixh_ms * 2 : h < 18 ? day_start + sixh_ms * 3 : day_start + sixh_ms * 4
    out

f_open_at_ts(int ts) =>
    float out = na

f_find_tcisd_anchor_ts(int start_ts, bool want_bull_candle, array<int> tArr, array<float> oArr, array<float> cArr) =>
    int out = na
    int n = array.size(tArr)
    if not na(start_ts) and n > 0
        for j = 0 to n - 1
            int idx = n - 1 - j
            int t0 = array.get(tArr, idx)
            if t0 <= start_ts
                float o0 = array.get(oArr, idx)
                float c0 = array.get(cArr, idx)
                bool color_ok = want_bull_candle ? c0 > o0 : c0 < o0
                if color_ok
                    out := t0
                    break
    out
