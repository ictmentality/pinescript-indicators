f_qt_day_start_raw(int t, string tz, int dayStartHour, int dayStartMin) =>
    int y  = year(t, tz)
    int mo = month(t, tz)
    int da = dayofmonth(t, tz)
    int h  = hour(t, tz)
    int mi = minute(t, tz)
    int base = timestamp(tz, y, mo, da, dayStartHour, dayStartMin)
    int out  = (h > dayStartHour or (h == dayStartHour and mi >= dayStartMin)) ? base : base - LEN_24H
    out

f_qt_day_start(int t) =>
    f_qt_day_start_raw(t, TZ, 18, 0)

f_next_ds_after(int t) =>
    int ds0 = f_qt_day_start(t)
    ds0 <= t ? ds0 + LEN_24H : ds0

f_ceil_to_1h(int ts) =>
    int step = 60 * MS_MIN
    int rem  = ts % step
    rem == 0 ? ts : ts + (step - rem)

f_weekly_aligned_open_expr(int q1Ts, int q2Ts, int q3Ts, int q4Ts, int q5Ts, int endTs) =>
    int tRef = time
    bool inW = tRef >= q1Ts and tRef < endTs

    int qi = 1
    if inW
        if tRef >= q2Ts
            qi := 2
        if tRef >= q3Ts
            qi := 3
        if tRef >= q4Ts
            qi := 4
        if tRef >= q5Ts
            qi := 5

    int qCurTs = int(na)
    if inW
        qCurTs := qi == 1 ? q1Ts : qi == 2 ? q2Ts : qi == 3 ? q3Ts : qi == 4 ? q4Ts : q5Ts

    int targetTs = na(qCurTs) ? int(na) : f_ceil_to_1h(qCurTs)

    var int   lastTarget = int(na)
    var float actualOpen = float(na)
    var int   actualOpenTs = int(na)
    var int   actualBoundaryTs = int(na)
    var int   actualQi = int(na)

    if na(lastTarget) or targetTs != lastTarget
        lastTarget := targetTs
        actualOpen := float(na)
        actualOpenTs := int(na)
        actualBoundaryTs := int(na)
        actualQi := int(na)

    bool hit = not na(targetTs) and (time >= targetTs) and (na(time[1]) or time[1] < targetTs)
    if hit and na(actualOpen)
        actualOpen := open
        actualOpenTs := time
        actualBoundaryTs := qCurTs
        actualQi := qi

    [actualOpen, actualOpenTs, actualBoundaryTs, actualQi]

f_weekly_aligned_open_current(int q1Ts, int q2Ts, int q3Ts, int q4Ts, int q5Ts, int endTs) =>
    request.security(

        "60",
        f_weekly_aligned_open_expr(q1Ts, q2Ts, q3Ts, q4Ts, q5Ts, endTs),
        lookahead=barmerge.lookahead_off
    )

    string s = str.upper(sym)
    string[] parts = str.split(s, ":")
    array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : s

f_check_match(string input_sym) =>
    bool valid = str.length(input_sym) > 0 and str.contains(input_sym, ":")
    if not valid
        false
    else
        string inputUp = str.upper(input_sym)
        string[] parts = str.split(inputUp, ":")
        string tSym = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : inputUp
        string curTicker = str.upper(syminfo.ticker)
        string curId = str.upper(syminfo.tickerid)
        (curTicker == tSym) or str.contains(curId, tSym) or str.contains(inputUp, curTicker)

f_ssmt_tag(string sym) =>
    string s = f_ssmt_sym_ticker(sym)
    string tag = s
    if str.contains(s, "MES")
        tag := "ES"
    else if str.contains(s, "MNQ")
        tag := "NQ"
    else if str.contains(s, "MYM")
        tag := "YM"
    else if str.contains(s, "ES")
        tag := "ES"
    else if str.contains(s, "NQ")
        tag := "NQ"
    else if str.contains(s, "YM")
        tag := "YM"
    tag

f_ssmt_weekly_highs_expr() =>
    int wkStart = time("W")
    int wkEnd = wkStart + 7 * LEN_24H
    int q2 = f_next_ds_after(wkStart)
    int q3 = q2 + LEN_24H
    int q4 = q3 + LEN_24H
    int q5 = q4 + LEN_24H

    int qCur = wkStart
    if time >= q5 and time < wkEnd
        qCur := q5
    else if time >= q4 and time < wkEnd
        qCur := q4
    else if time >= q3 and time < wkEnd
        qCur := q3
    else if time >= q2 and time < wkEnd
        qCur := q2

    var int qLast = int(na)
    var float hCur = float(na)
    var int hCurTs = int(na)
    var float hPrev = float(na)
    var int hPrevTs = int(na)
    var float bhCur = float(na)
    var int bhCurTs = int(na)
    var float bhPrev = float(na)
    var int bhPrevTs = int(na)
    float bodyHigh = math.max(open, close)

    if na(qLast) or qCur != qLast
        hPrev := hCur
        hPrevTs := hCurTs
        hCur := high
        hCurTs := time
        bhPrev := bhCur
        bhPrevTs := bhCurTs
        bhCur := bodyHigh
        bhCurTs := time
        qLast := qCur
    else
        if na(hCur) or high > hCur
            hCur := high
            hCurTs := time
        if na(bhCur) or bodyHigh > bhCur
            bhCur := bodyHigh
            bhCurTs := time

    [hCur, hCurTs, hPrev, hPrevTs, bhCur, bhCurTs, bhPrev, bhPrevTs, qCur]

f_tpd_live_sym(float _xc0, float _xo0, float _xh0, float _xl0, int _xt0, string _tf) =>
    var float _c1 = 0.0

    var int _t1 = 0
    var float _c2 = 0.0
    var float _o2 = 0.0
    var float _h2 = 0.0
    var float _l2 = 0.0
    var int _t2 = 0
    var bool up0 = false
    var bool up1 = false
    var bool dw0 = false
    var bool dw1 = false
    var int _l0t = time
    var int _h0t = time
    var int _h1t = time
    var int _l1t = time
    var float _o0 = open
    var float _h0 = high
    var float _l0 = low
    var int _t0 = time
    var float _c0 = close
    if timeframe.change(_tf)
        _t0 := _xt0
        _o0 := _xo0
        _h0 := _xh0
        _l0 := _xl0

    if _xh0 >= _h0
        _h0 := _xh0
        _h0t := time
    if _xl0 <= _l0
        _l0 := _xl0
        _l0t := time
    _c0 := _xc0
    up0 := _c0 > _o0
    dw0 := _c0 < _o0
    if timeframe.change(_tf)
        _c1 := _c0[1]
        _o1 := _o0[1]
        _h1 := _h0[1]
        _l1 := _l0[1]
        _t1 := _t0[1]
        _c2 := _c1[1]
        _o2 := _o1[1]
        _h2 := _h1[1]
        _l2 := _l1[1]
        _t2 := _t1[1]
        up1 := up0[1]
        dw1 := dw0[1]
        _l1t := _l0t[1]
        _h1t := _h0t[1]

    [_h2, _l2, _c2, _o2, _t2, _h1, _l1, _c1, _o1, _t1, _h0, _l0, _c0, _o0, _t0, up1, dw1, _l1t, _h1t]

f_tpd_tf_text(string _tf) =>
    float _tf_num = str.tonumber(_tf)
    int _tf_int = int(na)
    string _text = _tf
    if not na(_tf_num)
        _tf_int := int(_tf_num)
    if _tf == "1D"
        _text := "D"
    else if _tf == "1W"
        _text := "W"
    else if _tf == "1M"
        _text := "M"
    else if not na(_tf_int)
        if _tf_int < 60
            _text := "m" + _tf
        else if _tf_int % 60 == 0
            _text := "H" + str.tostring(_tf_int / 60)
        else
            _text := "m" + _tf
    _text

f_tpd_short_sym(string _s) =>
    string out = _s
    if str.length(out) == 0
        out := ""
    else if str.contains(out, "replay-feed")
        out := syminfo.ticker
    else if str.contains(out, ":")
        out := array.get(str.split(out, ":"), 1)
    out

f_tpd_fmt(float _v) =>

f_tpd_smt_pair(float _lo_c2_a, float _lo_c3_a, float _hi_c2_a, float _hi_c3_a, float _lo_c2_b, float _lo_c3_b, float _hi_c2_b, float _hi_c3_b) =>
    bool a_low_sweep = not na(_lo_c2_a) and not na(_lo_c3_a) and _lo_c3_a < _lo_c2_a
    bool b_low_sweep = not na(_lo_c2_b) and not na(_lo_c3_b) and _lo_c3_b < _lo_c2_b
    bool a_high_sweep = not na(_hi_c2_a) and not na(_hi_c3_a) and _hi_c3_a > _hi_c2_a
    bool b_high_sweep = not na(_hi_c2_b) and not na(_hi_c3_b) and _hi_c3_b > _hi_c2_b
    bool bull_smt = (a_low_sweep and not b_low_sweep) or (b_low_sweep and not a_low_sweep)
    bool bear_smt = (a_high_sweep and not b_high_sweep) or (b_high_sweep and not a_high_sweep)
    [bull_smt, bear_smt]

f_week_n(int[] startsIn, int[] endsIn, float[] woIn, float[] twoIn) =>
    int nS = array.size(startsIn)
    int nE = array.size(endsIn)
    int nW = array.size(woIn)
    int nT = array.size(twoIn)
    int n0 = nS < nE ? nS : nE
    int n1 = nW < nT ? nW : nT
    n0 < n1 ? n0 : n1

f_weekly_scan_update(
    WeeklyState stIn,
    WeeklyConfig cfg,
    int t,
    int tClose,
    int wkStartTs,
    int wkEndProjTs,
    float wkOpen,
    bool enabled
) =>
    int[] starts2 = array.copy(stIn.starts)
    int[] ends2 = array.copy(stIn.ends)
    float[] wo2 = array.copy(stIn.wo)
    float[] two2 = array.copy(stIn.two)
    float[] three2 = array.copy(stIn.three)
    float[] four2 = array.copy(stIn.four)
    float[] five2 = array.copy(stIn.five)
    int[] woTs2 = array.copy(stIn.woTs)
    int[] twoTs2 = array.copy(stIn.twoTs)
    int[] threeTs2 = array.copy(stIn.threeTs)
    int[] fourTs2 = array.copy(stIn.fourTs)
    int[] fiveTs2 = array.copy(stIn.fiveTs)
    int lastStart2 = stIn.lastStartTs

    WeeklyResult res = f_weekly_result_empty()

    if enabled and not na(wkStartTs)
        bool cycleChanged = na(lastStart2) or wkStartTs != lastStart2

        if cycleChanged
            int nEnds = array.size(ends2)
            if bar_index > 0 and nEnds > 0
                array.set(ends2, nEnds - 1, time_close[1])

            array.push(starts2, wkStartTs)
            array.push(ends2, int(na))
            array.push(wo2, wkOpen)
            array.push(two2, float(na))
            array.push(three2, float(na))
            array.push(four2, float(na))
            array.push(five2, float(na))
            array.push(woTs2, wkStartTs)
            array.push(twoTs2, int(na))
            array.push(threeTs2, int(na))
            array.push(fourTs2, int(na))
            array.push(fiveTs2, int(na))
            lastStart2 := wkStartTs

            while array.size(starts2) > cfg.keepCycles
                array.shift(starts2)
                array.shift(ends2)
                array.shift(wo2)
                array.shift(two2)
                array.shift(three2)

type vset
    line[] ln
    int[]  ts

vset_new() =>
    vset.new(array.new_line(), array.new_int())

vset_clear(vset s) =>
    int i = s.ln.size() - 1
    while i >= 0
        line.delete(s.ln.get(i))
        i -= 1
    s.ln.clear()
    s.ts.clear()
    0

vset_trim_since(vset s, int cutoff) =>
    int i = s.ts.size() - 1
    while i >= 0
        if s.ts.get(i) < cutoff
            line.delete(s.ln.get(i))
            array.remove(s.ln, i)
            array.remove(s.ts, i)
        i -= 1
    0

vset_has(vset s, int ts) =>
    bool found = false
    int i = s.ts.size() - 1
    while i >= 0 and not found
        found := s.ts.get(i) == ts
        i -= 1
    found

vset_add(vset s, int ts, color c) =>
    if not vset_has(s, ts)
        s.ln.push(line.new(ts, low, ts, high, xloc=xloc.bar_time, extend=extend.both, style=line.style_solid, width=1, color=f_vcol(c)))
        s.ts.push(ts)
    0

type hset
    line[] ln
    int[]  st

hset_new() =>
    hset.new(array.new_line(), array.new_int())

hset_clear(hset s) =>
    int i = s.ln.size() - 1
    while i >= 0
        line.delete(s.ln.get(i))
        i -= 1
    s.ln.clear()
    s.st.clear()
    0

hset_trim_since(hset s, int cutoff) =>
    int i = s.st.size() - 1
    while i >= 0
        if s.st.get(i) < cutoff
            line.delete(s.ln.get(i))
            array.remove(s.ln, i)
            array.remove(s.st, i)
        i -= 1
    0

hset_has(hset s, int st) =>
    bool found = false
    int i = s.st.size() - 1
    while i >= 0 and not found
        found := s.st.get(i) == st
        i -= 1
    found

hset_add(hset s, int st, int et, float y, color c, string styleStr) =>
    if not hset_has(s, st)
        s.ln.push(line.new(st, y, et, y, xloc=xloc.bar_time, extend=extend.none, style=lineStyleFrom(styleStr), width=1, color=c))
        s.st.push(st)
    0

hset_sync(hset s, color c, string styleStr) =>
    int i = s.ln.size() - 1
    while i >= 0
        line ln = s.ln.get(i)
        if not na(ln)
            line.set_color(ln, c)
            line.set_style(ln, lineStyleFrom(styleStr))
            line.set_width(ln, 1)
        i -= 1
    0

f_ssmt_hist_push(line ln, label lb, line[] lines, label[] labels, int maxN) =>
    if not na(ln) or not na(lb)
        if maxN <= 0
            if not na(ln)
                line.delete(ln)
            if not na(lb)
                label.delete(lb)
        else
            array.push(lines, ln)
            array.push(labels, lb)
            while array.size(lines) > maxN
                line oldLn = array.shift(lines)
                if not na(oldLn)
                    line.delete(oldLn)
                label oldLb = array.shift(labels)
                if not na(oldLb)
                    label.delete(oldLb)
    0

f_ssmt_hist_trim(line[] lines, label[] labels, int maxN) =>
    while array.size(lines) > maxN
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    while array.size(labels) > maxN
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

f_ssmt_hist_clear(line[] lines, label[] labels) =>
    while array.size(lines) > 0
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
    while array.size(labels) > 0
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

f_kill(line ln) =>
    line out = na
    if not na(ln)
        line.delete(ln)
    out

// WEEKLY CYCLE RUNNER
// ============================================================================
run_week_cycle(
    int[]   wkStarts,
    int[]   wkEnds,
    float[] wkWo,
    float[] wkTwo,
    int     wEndProj,
    int     keepWeeks,
    vset    v,
    hset    hWO,
    hset    hTWO,
    line    projQ2,
    line    projQ3,
    line    projQ4,
    line    projQ5,
    line    projEnd
) =>
    line p2  = projQ2
    line p3  = projQ3
    line p4  = projQ4
    line p5  = projQ5
    line pE  = projEnd

    int nS = wkStarts.size()
    int nE = wkEnds.size()
    int nW = wkWo.size()
    int nT = wkTwo.size()
    int n0 = nS < nE ? nS : nE
    int n1 = nW < nT ? nW : nT
    int n  = n0 < n1 ? n0 : n1

    if n <= 0
        p2 := f_kill(p2)
        p3 := f_kill(p3)
        p4 := f_kill(p4)
        p5 := f_kill(p5)
        pE := f_kill(pE)
        line[] out0 = array.new_line()
        array.push(out0, p2)
        array.push(out0, p3)
        array.push(out0, p4)
        array.push(out0, p5)
        array.push(out0, pE)
        out0
    else
        int first = math.max(0, n - keepWeeks)
        int cut   = wkStarts.get(first)

        vset_trim_since(v, cut)
        if showOpens
            hset_trim_since(hWO, cut)
            hset_trim_since(hTWO, cut)

        int i = first
        while i < n
            int wS = wkStarts.get(i)
            int wEStored = wkEnds.get(i)
            int wE = na(wEStored) ? wEndProj : wEStored

            if wS >= cut and time_close > wS
                vset_add(v, wS, COL_Q1)

            int q2 = f_next_ds_after(wS)
            int q3 = q2 + LEN_24H
            int q4 = q3 + LEN_24H
            int q5 = q4 + LEN_24H

            if not na(q2) and q2 >= cut and q2 < wE and time_close > q2
                vset_add(v, q2, COL_Q2)
            if not na(q3) and q3 >= cut and q3 < wE and time_close > q3
                vset_add(v, q3, COL_Q3)
            if not na(q4) and q4 >= cut and q4 < wE and time_close > q4
                vset_add(v, q4, COL_Q4)
            if not na(q5) and q5 >= cut and q5 < wE and time_close > q5
                vset_add(v, q5, COL_Q5)

            if not na(wEStored) and wEStored >= cut and time_close > wEStored
                vset_add(v, wEStored, COL_Q1)

            if showOpens
                float wo = wkWo.get(i)
                if not na(wo) and time_close > wS
                    hset_add(hWO, wS, wE, wo, q1OpenCol, q1OpenStyleStr)

                float twoO = wkTwo.get(i)
                if not na(twoO) and time_close > q2
                    hset_add(hTWO, q2, wE, twoO, trueOpenCol, trueOpenStyleStr)

            i += 1

        if showOpens
            hset_sync(hWO, q1OpenCol, q1OpenStyleStr)
            hset_sync(hTWO, trueOpenCol, trueOpenStyleStr)

        int curWs = wkStarts.get(n - 1)
        int curQ2 = f_next_ds_after(curWs)
        int curQ3 = curQ2 + LEN_24H
        int curQ4 = curQ3 + LEN_24H
        int curQ5 = curQ4 + LEN_24H

        if not na(curQ2) and time_close <= curQ2
            p2 := f_set_or_make_vline(p2, curQ2, COL_Q2)
        else
            p2 := f_kill(p2)

        if not na(curQ3) and time_close <= curQ3
            p3 := f_set_or_make_vline(p3, curQ3, COL_Q3)
        else
            p3 := f_kill(p3)

        if not na(curQ4) and time_close <= curQ4
            p4 := f_set_or_make_vline(p4, curQ4, COL_Q4)
        else
            p4 := f_kill(p4)

        if not na(curQ5) and time_close <= curQ5
            p5 := f_set_or_make_vline(p5, curQ5, COL_Q5)
        else
            p5 := f_kill(p5)

        pE := f_set_or_make_vline(pE, wEndProj, COL_Q1)

        line[] out = array.new_line()
        array.push(out, p2)
        array.push(out, p3)
        array.push(out, p4)
        array.push(out, p5)
        array.push(out, pE)
        out
