// ============================================================================
// QT Engine — Fixed + Elastic cycles (Daily / m90 / Micro / Nano / Weekly / Monthly)
// - Data-first: timestamps + OHLC + state flags
// - No drawings, no styling opinions
// - Timestamp anchoring only (bar_time), never bar_index
// - State-trigger correctness everywhere: realized = (time_close > ts)
// - Non-repaint: any request.security(..., lookahead_off) only
// - Elastic cycles (weekly/monthly) are implemented here
//
// NOTE: Micro/Nano use aligned-TF quarters for H/L/C with aligned actual-open.
//       90m/Daily use aligned-TF quarter OHLC with aligned actual-open.
// ============================================================================

f_q_idx_from_boundary(int boundaryTs, int q1Ts, int q2Ts, int q3Ts, int q4Ts, int q5Ts) =>
    int idx = 0
    if not na(boundaryTs)
        if boundaryTs == q1Ts
            idx := 1
        else if boundaryTs == q2Ts
            idx := 2
        else if boundaryTs == q3Ts
            idx := 3
        else if boundaryTs == q4Ts
            idx := 4
        else if boundaryTs == q5Ts
            idx := 5
    idx

f_qt_event_none() =>
    QtBoundaryEvent.new(false, int(na), int(na))

f_qt_boundary_event(FixedCycleState prev, FixedCycleState cur, CycleResult res) =>
    bool happened = not na(cur.lastBoundaryTs) and (na(prev.lastBoundaryTs) or cur.lastBoundaryTs != prev.lastBoundaryTs)
    int boundaryTs = happened ? cur.lastBoundaryTs : int(na)
    int qi = happened ? f_q_idx_from_boundary(boundaryTs, res.q1Ts, res.q2Ts, res.q3Ts, res.q4Ts, res.q5Ts) : int(na)
    QtBoundaryEvent.new(happened, boundaryTs, qi)

f_qt_day_start_raw(int t, string tz, int dayStartHour, int dayStartMin) =>
    int y  = year(t, tz)
    int mo = month(t, tz)
    int da = dayofmonth(t, tz)
    int h  = hour(t, tz)
    int mi = minute(t, tz)
    int base = timestamp(tz, y, mo, da, dayStartHour, dayStartMin)
    int out  = (h > dayStartHour or (h == dayStartHour and mi >= dayStartMin)) ? base : base - LEN_24H
    out

f_qt_day_start(int t, QTConfig cfg) =>
    f_qt_day_start_raw(t, cfg.tz, cfg.dayStartHour, cfg.dayStartMin)

f_next_ds_after(int t, QTConfig cfg) =>
    int ds0 = f_qt_day_start(t, cfg)
    ds0 <= t ? ds0 + LEN_24H : ds0

f_shift_ds_days(int t, QTConfig cfg, int days) =>
    int y = year(t, cfg.tz)
    int mo = month(t, cfg.tz)
    int da = dayofmonth(t, cfg.tz)
    timestamp(cfg.tz, y, mo, da + days, cfg.dayStartHour, cfg.dayStartMin)

f_month_week_start(int tRef, QTConfig cfg) =>
    int ds = f_qt_day_start(tRef, cfg)
    int dow = dayofweek(ds, cfg.tz)
    int back = dow == dayofweek.sunday ? 0 : dow == dayofweek.monday ? 1 : dow == dayofweek.tuesday ? 2 : dow == dayofweek.wednesday ? 3 : dow == dayofweek.thursday ? 4 : dow == dayofweek.friday ? 5 : 6
    f_shift_ds_days(ds, cfg, -back)

f_month_week_key(int weekStartTs, QTConfig cfg) =>
    int mondayTs = f_shift_ds_days(weekStartTs, cfg, 1)
    int y = year(mondayTs, cfg.tz)
    int mo = month(mondayTs, cfg.tz)
    y * 100 + mo

f_monthly_cycle_bounds(int tRef, QTConfig cfg) =>
    int wsCur = f_month_week_start(tRef, cfg)
    int keyCur = f_month_week_key(wsCur, cfg)

    int probeBack = f_shift_ds_days(wsCur, cfg, -7)
    int guardBack = 0
    while guardBack < 8 and f_month_week_key(probeBack, cfg) == keyCur
        probeBack := f_shift_ds_days(probeBack, cfg, -7)
        guardBack += 1
    int cStart = f_shift_ds_days(probeBack, cfg, 7)

    int probeFwd = f_shift_ds_days(wsCur, cfg, 7)
    int guardFwd = 0
    while guardFwd < 8 and f_month_week_key(probeFwd, cfg) == keyCur
        probeFwd := f_shift_ds_days(probeFwd, cfg, 7)
        guardFwd += 1
    int cEnd = probeFwd

    int q1Ts = cStart
    int q2Ts = int(na)
    int q3Ts = int(na)
    int q4Ts = int(na)
    int q5Ts = int(na)

    int b2 = f_shift_ds_days(q1Ts, cfg, 7)
    if b2 < cEnd
        q2Ts := b2
    if not na(q2Ts)
        int b3 = f_shift_ds_days(q2Ts, cfg, 7)
        if b3 < cEnd
            q3Ts := b3
    if not na(q3Ts)
        int b4 = f_shift_ds_days(q3Ts, cfg, 7)
        if b4 < cEnd
            q4Ts := b4
    if not na(q4Ts)
        int b5 = f_shift_ds_days(q4Ts, cfg, 7)
        if b5 < cEnd
            q5Ts := b5

    [cStart, cEnd, q1Ts, q2Ts, q3Ts, q4Ts, q5Ts]

f_cycle_start(int t, int baseTs, int lenMs) =>
    // Computes start = baseTs + floor((t - baseTs)/lenMs) * lenMs
    // Assumes baseTs is the correct anchor for t (e.g., DS for the day containing t)
    int k = int(math.floor((t - baseTs) / lenMs))
    baseTs + k * lenMs

f_quarter_index(int t, int cycleStart, int lenMs) =>
    int qLen = int(lenMs / 4)
    int qi0  = int(math.floor((t - cycleStart) / qLen))  // 0..3
    int qi1  = qi0 < 0 ? 0 : (qi0 > 3 ? 3 : qi0)
    qi1 + 1

// CYCLE WINDOW MODULE (timestamp math only)
// - Fixed lengths for Daily/m90/Micro/Nano
// - Elastic bounds for Weekly/Monthly
// ============================================================================

f_fixed_cycle_window(int tRef, QTConfig cfg, int lenMs) =>
    bool isWeeklyElastic = lenMs == LEN_WEEK_ELASTIC
    bool isMonthlyElastic = lenMs == LEN_MONTH_ELASTIC
    int cStart = isWeeklyElastic ? time("W") : int(na)
    int cEnd = isWeeklyElastic ? time_close("W") : int(na)
    if not isWeeklyElastic and not isMonthlyElastic
        int ds = f_qt_day_start(tRef, cfg)
        cStart := f_cycle_start(tRef, ds, lenMs)
        cEnd := cStart + lenMs

    int q1Ts = cStart
    int q2Ts = int(na)
    int q3Ts = int(na)
    int q4Ts = int(na)
    int q5Ts = int(na)

    if isWeeklyElastic
        q2Ts := f_next_ds_after(q1Ts, cfg)
        q3Ts := q2Ts + LEN_24H
        q4Ts := q3Ts + LEN_24H
        q5Ts := q4Ts + LEN_24H
    else if isMonthlyElastic
        [mStart, mEnd, mQ1, mQ2, mQ3, mQ4, mQ5] = f_monthly_cycle_bounds(tRef, cfg)
        cStart := mStart
        cEnd := mEnd
        q1Ts := mQ1
        q2Ts := mQ2
        q3Ts := mQ3
        q4Ts := mQ4
        q5Ts := mQ5
    else
        int span = cEnd - cStart
        int qLen = int(math.floor(span / 4))
        if na(qLen) or qLen <= 0
            qLen := 1
        q2Ts := cStart + 1 * qLen
        q3Ts := cStart + 2 * qLen
        q4Ts := cStart + 3 * qLen

    int qi = 1
    bool hasQ5 = not na(q5Ts)
    if hasQ5 and tRef >= q5Ts and tRef < cEnd
        qi := 5
    else if tRef >= q4Ts and tRef < cEnd
        qi := 4
    else if tRef >= q3Ts and tRef < cEnd
        qi := 3
    else if tRef >= q2Ts and tRef < cEnd
        qi := 2
    else if lenMs > 0
        qi := f_quarter_index(tRef, cStart, lenMs)
    bool inW = tRef >= cStart and tRef < cEnd
    [cStart, cEnd, q1Ts, q2Ts, q3Ts, q4Ts, q5Ts, qi, inW]

f_fixed_cycle_boundary_ts(int q1Ts, int q2Ts, int q3Ts, int q4Ts, int q5Ts, int qi) =>
    int qCurTs = q1Ts
    if qi == 2
        qCurTs := q2Ts
    else if qi == 3
        qCurTs := q3Ts
    else if qi == 4
        qCurTs := q4Ts
    else if qi == 5
        qCurTs := q5Ts
    qCurTs

f_fixed_cycle_next_boundary_ts(int q1Ts, int q2Ts, int q3Ts, int q4Ts, int q5Ts, int qi, bool inW) =>
    int qCurTs = f_fixed_cycle_boundary_ts(q1Ts, q2Ts, q3Ts, q4Ts, q5Ts, qi)
    inW ? qCurTs : int(na)

f_fixed_cycle_quarter_id(int boundaryTs, int q1Ts, int q2Ts, int q3Ts, int q4Ts, int q5Ts) =>
    f_q_idx_from_boundary(boundaryTs, q1Ts, q2Ts, q3Ts, q4Ts, q5Ts)

f_ceil_to_5s(int ts) =>
    int step = 5 * MS_SEC
    int rem  = ts % step
    rem == 0 ? ts : ts + (step - rem)

f_ceil_to_30s(int ts) =>
    int step = 30 * MS_SEC
    int rem  = ts % step
    rem == 0 ? ts : ts + (step - rem)

f_ceil_to_5m(int ts) =>
    int step = 5 * MS_MIN
    int rem  = ts % step
    rem == 0 ? ts : ts + (step - rem)

f_ceil_to_15m(int ts) =>
    int step = 15 * MS_MIN
    int rem  = ts % step
    rem == 0 ? ts : ts + (step - rem)

f_micro_aligned_ohlc_expr() =>
    int tRef = time
    int ds = f_qt_day_start_raw(tRef, NANO_TZ, NANO_DAY_START_HOUR, NANO_DAY_START_MIN)

    int cStart = f_cycle_start(tRef, ds, LEN_90M)
    int qLen   = int(LEN_90M / 4)

    int q1Ts = cStart
    int q2Ts = cStart + 1 * qLen
    int q3Ts = cStart + 2 * qLen
    int q4Ts = cStart + 3 * qLen

    int qi = f_quarter_index(tRef, cStart, LEN_90M)

    int qCurTs = q1Ts
    if qi == 2
        qCurTs := q2Ts
    else if qi == 3
        qCurTs := q3Ts
    else if qi == 4
        qCurTs := q4Ts

    int targetTs = f_ceil_to_30s(qCurTs)

    var int   lastTarget = int(na)
    var float actualOpen = float(na)
    var int   actualOpenTs = int(na)
    var int   actualBoundaryTs = int(na)
    var int   actualQi = int(na)

    if na(lastTarget) or targetTs != lastTarget
        lastTarget := targetTs
        actualOpen := float(na)
        actualOpenTs := int(na)
        actualBoundaryTs := int(na)
        actualQi := int(na)

    bool hit = (time >= targetTs) and (na(time[1]) or time[1] < targetTs)
    if hit and na(actualOpen)
        actualOpen := open
        actualOpenTs := time
        actualBoundaryTs := qCurTs
        actualQi := qi

    [open, high, low, close, time, time_close, actualOpen, actualOpenTs, actualBoundaryTs, actualQi]

f_micro_aligned_ohlc_current() =>
    request.security(
        syminfo.tickerid,
        MICRO_OPEN_TF,
        f_micro_aligned_ohlc_expr(),
        lookahead=barmerge.lookahead_off
    )

    if hit and na(actualOpen)
        actualOpen := open
        actualOpenTs := time
        actualBoundaryTs := qCurTs
        actualQi := qi

    [open, high, low, close, time, time_close, actualOpen, actualOpenTs, actualBoundaryTs, actualQi]

f_q_clear_evts(QuarterRecord qIn) =>
    QuarterRecord q = qIn
    q.evtStart := false
    q.evtFinalize := false
    q

f_q_finalize(QuarterRecord qIn) =>
    QuarterRecord q = qIn
    q.isFinal := true
    q.evtFinalize := true
    if not q.has
        q.isEmpty := true
    q

f_finalize_and_roll(QuarterRecord qIn, QuarterRecord prev1In, QuarterRecord prev2In) =>
    QuarterRecord q = qIn
    QuarterRecord p1 = prev1In
    QuarterRecord p2 = prev2In
    if not q.isFinal and not na(q.boundaryTs)
        q := f_q_finalize(q)
        // Keep prev buffers on last finalized quarter with actual data.
        if q.has
            p2 := p1
            p1 := q
    [q, p1, p2]

f_q_apply_bar(QuarterRecord qIn, int t, float o, float h, float l, float c, bool isStartBar) =>
    QuarterRecord q = qIn
    float barBH = math.max(o, c)
    float barBL = math.min(o, c)

    if not q.has
        q.o := o
        q.oTs := t
        q.h := h
        q.hTs := t
        q.l := l
        q.lTs := t
        q.c := c
        q.cTs := t
        q.bh := barBH
        q.bl := barBL
        q.bhTs := t
        q.blTs := t
        q.has := true
        q.isEmpty := false
        q.isFinal := false
        q.evtStart := isStartBar
    else
        bool newHigh = na(q.h) or h > q.h
        bool newLow  = na(q.l) or l < q.l
        bool newBH   = na(q.bh) or barBH > q.bh
        bool newBL   = na(q.bl) or barBL < q.bl
        if newHigh
            q.h := h
            q.hTs := t
        if newLow
            q.l := l
            q.lTs := t
        if newBH
            q.bh := barBH
            q.bhTs := t
        if newBL
            q.bl := barBL
            q.blTs := t
        q.c := c
        q.cTs := t
        if isStartBar
            q.evtStart := true

    q

f_apply_aligned_qr(QuarterRecord qIn, QuarterRecord aIn) =>
    QuarterRecord q = qIn
    q.h := aIn.h
    q.hTs := aIn.hTs
    q.l := aIn.l
    q.lTs := aIn.lTs
    q.c := aIn.c
    q.cTs := aIn.cTs
    q.bh := aIn.bh
    q.bhTs := aIn.bhTs
    q.bl := aIn.bl
    q.blTs := aIn.blTs
    q

f_core_make_event_if_realized(QtCycleState cs, int candidateBoundaryTs, int tClose) =>
    bool realized = not na(candidateBoundaryTs) and tClose > candidateBoundaryTs
    bool happened = realized and (na(cs.lastBoundaryTs) or candidateBoundaryTs != cs.lastBoundaryTs)
    int qi = happened ? f_q_idx_from_boundary(candidateBoundaryTs, cs.q1.boundaryTs, cs.q2.boundaryTs, cs.q3.boundaryTs, cs.q4.boundaryTs, cs.q5.boundaryTs) : int(na)
    QtBoundaryEvent.new(happened, happened ? candidateBoundaryTs : int(na), qi)

f_finalize_q_by_idx(
    int idx,
    QuarterRecord q1In,
    QuarterRecord q2In,
    QuarterRecord q3In,
    QuarterRecord q4In,
    QuarterRecord q5In,
    QuarterRecord prev1In,
    QuarterRecord prev2In
) =>
    QuarterRecord q1 = q1In
    QuarterRecord q2 = q2In
    QuarterRecord q3 = q3In
    QuarterRecord q4 = q4In
    QuarterRecord q5 = q5In
    QuarterRecord p1 = prev1In
    QuarterRecord p2 = prev2In

    if idx == 1
        [tq1, tp1, tp2] = f_finalize_and_roll(q1, p1, p2)
        q1 := tq1
        p1 := tp1
        p2 := tp2
    else if idx == 2
        [tq2, tp1, tp2] = f_finalize_and_roll(q2, p1, p2)
        q2 := tq2
        p1 := tp1
        p2 := tp2
    else if idx == 3
        [tq3, tp1, tp2] = f_finalize_and_roll(q3, p1, p2)
        q3 := tq3
        p1 := tp1
        p2 := tp2
    else if idx == 4
        [tq4, tp1, tp2] = f_finalize_and_roll(q4, p1, p2)
        q4 := tq4
        p1 := tp1
        p2 := tp2
    else if idx == 5
        [tq5, tp1, tp2] = f_finalize_and_roll(q5, p1, p2)
        q5 := tq5
        p1 := tp1
        p2 := tp2

    [q1, q2, q3, q4, q5, p1, p2]

f_core_roll_quarter(QtCycleState cs, QtBoundaryEvent evt) =>
    QtCycleState out = cs
    if evt.happened
        int lastIdx = f_q_idx_from_boundary(cs.lastBoundaryTs, cs.q1.boundaryTs, cs.q2.boundaryTs, cs.q3.boundaryTs, cs.q4.boundaryTs, cs.q5.boundaryTs)
        int newIdx = evt.quarterIndex

        QuarterRecord q1 = cs.q1
        QuarterRecord q2 = cs.q2
        QuarterRecord q3 = cs.q3
        QuarterRecord q4 = cs.q4
        QuarterRecord q5 = cs.q5
        QuarterRecord p1 = cs.prev1
        QuarterRecord p2 = cs.prev2

        int maxIdx = na(cs.q5.boundaryTs) ? 4 : 5
        int fromIdx = lastIdx == 0 ? 1 : lastIdx
        int toIdx = newIdx - 1
        if toIdx > maxIdx
            toIdx := maxIdx
        int idx = fromIdx
        while idx <= toIdx
            [rq1, rq2, rq3, rq4, rq5, rp1, rp2] = f_finalize_q_by_idx(idx, q1, q2, q3, q4, q5, p1, p2)
            q1 := rq1
            q2 := rq2
            q3 := rq3
            q4 := rq4
            q5 := rq5
            p1 := rp1
            p2 := rp2
            idx += 1

        out.q1 := q1
        out.q2 := q2
        out.q3 := q3
        out.q4 := q4
        out.q5 := q5
        out.prev1 := p1
        out.prev2 := p2
        out.lastBoundaryTs := evt.boundaryTs
        out.evt := evt
    else
        out.evt := f_qt_event_none()
    out

f_core_update_ohlc_at_idx(QtCycleState cs, int qIdx, int t, float o, float h, float l, float c, bool isStartBar) =>
    QtCycleState out = cs
    if qIdx == 1 and not cs.q1.isFinal
        out.q1 := f_q_apply_bar(cs.q1, t, o, h, l, c, isStartBar)
    else if qIdx == 2 and not cs.q2.isFinal
        out.q2 := f_q_apply_bar(cs.q2, t, o, h, l, c, isStartBar)
    else if qIdx == 3 and not cs.q3.isFinal
        out.q3 := f_q_apply_bar(cs.q3, t, o, h, l, c, isStartBar)
    else if qIdx == 4 and not cs.q4.isFinal
        out.q4 := f_q_apply_bar(cs.q4, t, o, h, l, c, isStartBar)
    else if qIdx == 5 and not cs.q5.isFinal
        out.q5 := f_q_apply_bar(cs.q5, t, o, h, l, c, isStartBar)
    out

f_core_update_ohlc(QtCycleState cs, int t, float o, float h, float l, float c, bool isStartBar) =>
    f_core_update_ohlc_at_idx(cs, cs.curQuarterIndex, t, o, h, l, c, isStartBar)

f_core_finalize_old_cycle(
    QuarterRecord q1In,
    QuarterRecord q2In,
    QuarterRecord q3In,
    QuarterRecord q4In,
    QuarterRecord q5In,
    QuarterRecord prev1In,
    QuarterRecord prev2In,
    int lastBoundaryTs
) =>
    QuarterRecord q1 = q1In
    QuarterRecord q2 = q2In
    QuarterRecord q3 = q3In
    QuarterRecord q4 = q4In
    QuarterRecord q5 = q5In
    QuarterRecord p1 = prev1In
    QuarterRecord p2 = prev2In

    int lastIdxOld = f_q_idx_from_boundary(lastBoundaryTs, q1.boundaryTs, q2.boundaryTs, q3.boundaryTs, q4.boundaryTs, q5.boundaryTs)
    int maxIdxOld = na(q5.boundaryTs) ? 4 : 5
    int fromIdx = lastIdxOld == 0 ? 1 : lastIdxOld
    int idx = fromIdx
    while idx <= maxIdxOld
        [rq1, rq2, rq3, rq4, rq5, rp1, rp2] = f_finalize_q_by_idx(idx, q1, q2, q3, q4, q5, p1, p2)
        q1 := rq1
        q2 := rq2
        q3 := rq3
        q4 := rq4
        q5 := rq5
        p1 := rp1
        p2 := rp2
        idx += 1

    [q1, q2, q3, q4, q5, p1, p2]


f_push_printed_start(int[] startsIn, int startTs, int keepCycles) =>
    // No mutation of input param: copy first, return the new array.
    int[] out = array.copy(startsIn)
    int sz = array.size(out)
    int last = sz > 0 ? array.get(out, sz - 1) : int(na)
    if na(last) or startTs > last
        array.push(out, startTs)

    // Cap to keepCycles (drop oldest)
    while array.size(out) > keepCycles
        array.shift(out)

    out

// CORE CYCLE UPDATE (Chart-TF aggregation + aligned overrides)
// - Enabled for Daily + m90 + Micro + Nano + Weekly + Monthly.
// - Aggregates quarter OHLC on chart TF into QuarterRecords, then applies
//   aligned-TF overrides for Micro/Nano (H/L/C) and m90/Daily (OHLC).
// - Gap-safe: skipped quarters finalize as isEmpty/isFinal on the next printed bar
// ============================================================================

f_fixed_cycle_update(
    FixedCycleState stIn,
    QTConfig cfg,
    int t,
    int tClose,
    int lenMs
) =>
    bool isNano = (lenMs == LEN_22M30S)
    bool isMicro = (lenMs == LEN_90M)
    bool isM90 = (lenMs == LEN_6H)
    bool isDaily = (lenMs == LEN_24H)
    bool isWeekly = (lenMs == LEN_WEEK_ELASTIC)
    bool isMonthly = (lenMs == LEN_MONTH_ELASTIC)
    int tRef = isNano ? t : (tClose - 1)

    [cStart, cEnd, q1Ts, q2Ts, q3Ts, q4Ts, q5Ts, qi, inW] = f_fixed_cycle_window(tRef, cfg, lenMs)
    bool hasQ5Cycle = (isWeekly or isMonthly) and not na(q5Ts)

    bool q2R = tClose > q2Ts
    bool q3R = tClose > q3Ts
    bool q4R = tClose > q4Ts
    bool q5R = hasQ5Cycle and tClose > q5Ts

    bool sR  = tClose > cStart
    bool eR  = tClose > cEnd

    // Printed start retention + cycle-change detection
    int[] starts2    = stIn.starts
    int  lastStart2  = stIn.lastStartTs
    bool cycleChanged = na(lastStart2) or cStart != lastStart2
    if cycleChanged
        starts2    := f_push_printed_start(stIn.starts, cStart, cfg.keepCycles)
        lastStart2 := cStart

    // Local copies (clear pulse events each bar)
    QuarterRecord q1 = f_q_clear_evts(stIn.q1)
    QuarterRecord q2 = f_q_clear_evts(stIn.q2)
    QuarterRecord q3 = f_q_clear_evts(stIn.q3)
    QuarterRecord q4 = f_q_clear_evts(stIn.q4)
    QuarterRecord q5 = f_q_clear_evts(stIn.q5)

    QuarterRecord prev1 = f_q_clear_evts(stIn.prev1)
    QuarterRecord prev2 = f_q_clear_evts(stIn.prev2)

    int lastBoundary2 = stIn.lastBoundaryTs
    int pendingBoundary2 = stIn.pendingBoundaryTs
    float pendingOpen2 = stIn.pendingOpen
    int pendingOpenTs2 = stIn.pendingOpenTs
    bool pendingHas2 = stIn.pendingHas
    QuarterRecord aq1 = f_q_clear_evts(stIn.aq1)
    QuarterRecord aq2 = f_q_clear_evts(stIn.aq2)
    QuarterRecord aq3 = f_q_clear_evts(stIn.aq3)
    QuarterRecord aq4 = f_q_clear_evts(stIn.aq4)
    QuarterRecord aq5 = f_q_clear_evts(stIn.aq5)
    QuarterRecord aprev1 = f_q_clear_evts(stIn.aprev1)
    QuarterRecord aprev2 = f_q_clear_evts(stIn.aprev2)
    int alignedLastStart2 = stIn.alignedLastStartTs
    int alignedLastBoundary2 = stIn.alignedLastBoundaryTs
    int alignedLastBar2 = stIn.alignedLastBarTs

    // If cycle rolled, finalize remaining old-cycle quarters (gap-safe), then reset records for the new cycle.
    if cycleChanged
        [tmpQ1, tmpQ2, tmpQ3, tmpQ4, tmpQ5, tmpP1, tmpP2] = f_core_finalize_old_cycle(q1, q2, q3, q4, q5, prev1, prev2, lastBoundary2)
        q1 := tmpQ1
        q2 := tmpQ2
        q3 := tmpQ3
        q4 := tmpQ4
        q5 := tmpQ5
        prev1 := tmpP1
        prev2 := tmpP2

        q1 := f_make_empty_q(q1Ts)
        q2 := f_make_empty_q(q2Ts)
        q3 := f_make_empty_q(q3Ts)
        q4 := f_make_empty_q(q4Ts)
        q5 := f_make_empty_q(q5Ts)
        lastBoundary2 := int(na)
        pendingBoundary2 := int(na)
        pendingOpen2 := float(na)
        pendingOpenTs2 := int(na)
        pendingHas2 := false

    // Ensure boundaries are initialized.
    // Weekly can carry event-driven boundaries that differ from fixed slot timestamps.
    if isWeekly
        if na(q1.boundaryTs)
            q1 := f_make_empty_q(q1Ts)
        if na(q2.boundaryTs)
            q2 := f_make_empty_q(q2Ts)
        if na(q3.boundaryTs)
            q3 := f_make_empty_q(q3Ts)
        if na(q4.boundaryTs)
            q4 := f_make_empty_q(q4Ts)
        if na(q5.boundaryTs)
            q5 := f_make_empty_q(q5Ts)
    else
        if na(q1.boundaryTs) or q1.boundaryTs != q1Ts
            q1 := f_make_empty_q(q1Ts)
        if na(q2.boundaryTs) or q2.boundaryTs != q2Ts
            q2 := f_make_empty_q(q2Ts)
        if na(q3.boundaryTs) or q3.boundaryTs != q3Ts
            q3 := f_make_empty_q(q3Ts)
        if na(q4.boundaryTs) or q4.boundaryTs != q4Ts
            q4 := f_make_empty_q(q4Ts)
        if na(q5.boundaryTs) or q5.boundaryTs != q5Ts
            q5 := f_make_empty_q(q5Ts)
    if pendingHas2 and pendingBoundary2 != q1Ts and pendingBoundary2 != q2Ts and pendingBoundary2 != q3Ts and pendingBoundary2 != q4Ts and pendingBoundary2 != q5Ts
        pendingBoundary2 := int(na)
        pendingOpen2 := float(na)
        pendingOpenTs2 := int(na)
        pendingHas2 := false

    // Current quarter boundary candidate for this bar
    int candidateBoundaryTs = f_fixed_cycle_next_boundary_ts(q1Ts, q2Ts, q3Ts, q4Ts, q5Ts, qi, inW)

    // Aligned-TF quarter aggregation (Micro/Nano/90m/Daily).
    bool useAligned = isM90 or isDaily
    [aO, aH, aL, aC, aT, aTClose, aActualOpen, aActualOpenTs, aActualBoundaryTs, aActualQi] = f_aligned_ohlc_pick(isNano, isMicro, isM90, isDaily)
    if not na(aActualOpen) and not na(aActualOpenTs) and not na(aActualBoundaryTs)
        if aActualBoundaryTs == q1Ts or aActualBoundaryTs == q2Ts or aActualBoundaryTs == q3Ts or aActualBoundaryTs == q4Ts or aActualBoundaryTs == q5Ts
            pendingBoundary2 := aActualBoundaryTs
            pendingOpen2 := aActualOpen
            pendingOpenTs2 := aActualOpenTs
            pendingHas2 := true

    bool aNew = not na(aT) and (na(alignedLastBar2) or aT != alignedLastBar2)
    if aNew
        alignedLastBar2 := aT
        int aRef = f_aligned_bar_ref(aTClose)
        [aCStart, aCEnd, aq1Ts, aq2Ts, aq3Ts, aq4Ts, aq5Ts, aQi, aInW] = f_fixed_cycle_window(aRef, cfg, lenMs)
        bool aCycleChanged = na(alignedLastStart2) or aCStart != alignedLastStart2
        if aCycleChanged
            [tmpAQ1, tmpAQ2, tmpAQ3, tmpAQ4, tmpAQ5, tmpAP1, tmpAP2] = f_core_finalize_old_cycle(aq1, aq2, aq3, aq4, aq5, aprev1, aprev2, alignedLastBoundary2)
            aq1 := tmpAQ1
            aq2 := tmpAQ2
            aq3 := tmpAQ3
            aq4 := tmpAQ4
            aq5 := tmpAQ5
            aprev1 := tmpAP1
            aprev2 := tmpAP2

            aq1 := f_make_empty_q(aq1Ts)
            aq2 := f_make_empty_q(aq2Ts)
            aq3 := f_make_empty_q(aq3Ts)
            aq4 := f_make_empty_q(aq4Ts)
            aq5 := f_make_empty_q(aq5Ts)
            alignedLastBoundary2 := int(na)
            alignedLastStart2 := aCStart

        if na(aq1.boundaryTs) or aq1.boundaryTs != aq1Ts
            aq1 := f_make_empty_q(aq1Ts)
        if na(aq2.boundaryTs) or aq2.boundaryTs != aq2Ts
            aq2 := f_make_empty_q(aq2Ts)
        if na(aq3.boundaryTs) or aq3.boundaryTs != aq3Ts
            aq3 := f_make_empty_q(aq3Ts)
        if na(aq4.boundaryTs) or aq4.boundaryTs != aq4Ts
            aq4 := f_make_empty_q(aq4Ts)
        if na(aq5.boundaryTs) or aq5.boundaryTs != aq5Ts
            aq5 := f_make_empty_q(aq5Ts)

        int aCandidateBoundaryTs = f_fixed_cycle_next_boundary_ts(aq1Ts, aq2Ts, aq3Ts, aq4Ts, aq5Ts, aQi, aInW)
        bool aQ2R = aTClose > aq2Ts
        bool aQ3R = aTClose > aq3Ts
        bool aQ4R = aTClose > aq4Ts
        bool aQ5R = (isWeekly or isMonthly) and not na(aq5Ts) and aTClose > aq5Ts
        bool aSR = aTClose > aCStart
        bool aER = aTClose > aCEnd

        QtCycleState aCore = QtCycleState.new(
            aCStart,
            aCEnd,
            aSR,
            aER,
            alignedLastBoundary2,
            aQi,
            aInW,
            aQ2R,
            aQ3R,
            aQ4R,
            aQ5R,
            aq1,
            aq2,
            aq3,
            aq4,
            aq5,
            aprev1,
            aprev2,
            f_qt_event_none(),
            0,
            0
        )

        QtBoundaryEvent aEvt = f_core_make_event_if_realized(aCore, aCandidateBoundaryTs, aTClose)
        QtCycleState aCore2 = f_core_roll_quarter(aCore, aEvt)
        if not na(aO) and not na(aH) and not na(aL) and not na(aC)
            aCore2 := f_core_update_ohlc(aCore2, aT, aO, aH, aL, aC, aEvt.happened)

        aq1 := aCore2.q1
        aq2 := aCore2.q2
        aq3 := aCore2.q3
        aq4 := aCore2.q4
        aq5 := aCore2.q5
        aprev1 := aCore2.prev1
        aprev2 := aCore2.prev2
        alignedLastBoundary2 := aCore2.lastBoundaryTs

    QtCycleState core = QtCycleState.new(
        cStart,
        cEnd,
        sR,
        eR,
        lastBoundary2,
        qi,
        inW,
        q2R,
        q3R,
        q4R,
        q5R,
        q1,
        q2,
        q3,
        q4,
        q5,
        prev1,
        prev2,
        f_qt_event_none(),
        0,
        0
    )

    QtBoundaryEvent evt = f_core_make_event_if_realized(core, candidateBoundaryTs, tClose)
    int qiOut = qi
    if isWeekly
        int hasCount = f_q_has_count(q1, q2, q3, q4, q5)
        int logicalIdx = hasCount > 0 ? hasCount : 1
        if evt.happened
            int nextIdx = hasCount + 1
            if nextIdx > 5
                nextIdx := 5
            evt := QtBoundaryEvent.new(true, evt.boundaryTs, nextIdx)
            logicalIdx := nextIdx
        qiOut := logicalIdx

    QtCycleState core2 = f_core_roll_quarter(core, evt)
    core2 := f_core_update_ohlc_at_idx(core2, qiOut, t, open, high, low, close, evt.happened)

    q1 := core2.q1
    q2 := core2.q2
    q3 := core2.q3
    q4 := core2.q4
    q5 := core2.q5
    prev1 := core2.prev1
    prev2 := core2.prev2
    lastBoundary2 := core2.lastBoundaryTs

    // Weekly boundary labels are event-driven: a started logical quarter keeps the observed anchor.
    if isWeekly and evt.happened
        if evt.quarterIndex == 1
            q1.boundaryTs := evt.boundaryTs
        else if evt.quarterIndex == 2
            q2.boundaryTs := evt.boundaryTs
        else if evt.quarterIndex == 3
            q3.boundaryTs := evt.boundaryTs
        else if evt.quarterIndex == 4
            q4.boundaryTs := evt.boundaryTs
        else if evt.quarterIndex == 5
            q5.boundaryTs := evt.boundaryTs

    // Phase 3: Apply aligned H/L/C/body values (opens preserved from pending actual-open).
    // NOTE: Actual open is preserved since aligned aggregation can miss sub-bars on higher chart TFs.
    bool applyAlignedValues = isNano or isMicro or useAligned
    if applyAlignedValues
        q1 := f_apply_aligned_qr(q1, aq1)
        q2 := f_apply_aligned_qr(q2, aq2)
        q3 := f_apply_aligned_qr(q3, aq3)
        q4 := f_apply_aligned_qr(q4, aq4)
        q5 := f_apply_aligned_qr(q5, aq5)

        // Align prev quarters' H/L/C/body values for 90m/Daily (keep chart/open from pending actual-open)
        if useAligned
            if aprev1.has
                prev1 := f_apply_aligned_qr(prev1, aprev1)
            if aprev2.has
                prev2 := f_apply_aligned_qr(prev2, aprev2)

    // Apply pending actual-open when its quarter becomes active.
    if pendingHas2
        if pendingBoundary2 == q1Ts and qiOut == 1
            q1.o   := pendingOpen2
            q1.oTs := pendingOpenTs2
            pendingHas2 := false
        else if pendingBoundary2 == q2Ts and qiOut == 2
            q2.o   := pendingOpen2
            q2.oTs := pendingOpenTs2
            pendingHas2 := false
        else if pendingBoundary2 == q3Ts and qiOut == 3
            q3.o   := pendingOpen2
            q3.oTs := pendingOpenTs2
            pendingHas2 := false
        else if pendingBoundary2 == q4Ts and qiOut == 4
            q4.o   := pendingOpen2
            q4.oTs := pendingOpenTs2
            pendingHas2 := false
        else if pendingBoundary2 == q5Ts and qiOut == 5
            q5.o   := pendingOpen2
            q5.oTs := pendingOpenTs2
            pendingHas2 := false
        else if (pendingBoundary2 == q1Ts and qiOut > 1) or (pendingBoundary2 == q2Ts and qiOut > 2) or (pendingBoundary2 == q3Ts and qiOut > 3) or (pendingBoundary2 == q4Ts and qiOut > 4)
            pendingBoundary2 := int(na)
            pendingOpen2 := float(na)
            pendingOpenTs2 := int(na)
            pendingHas2 := false

    bool q2ROut = q2R
    bool q3ROut = q3R
    bool q4ROut = q4R
    bool q5ROut = q5R
    if isWeekly
        q2ROut := q2R and q2.has
        q3ROut := q3R and q3.has
        q4ROut := q4R and q4.has
        q5ROut := q5R and q5.has

    FixedCycleState stOut = FixedCycleState.new(starts2, lastStart2, lastBoundary2, q1, q2, q3, q4, q5, prev1, prev2, pendingBoundary2, pendingOpen2, pendingOpenTs2, pendingHas2, aq1, aq2, aq3, aq4, aq5, aprev1, aprev2, alignedLastStart2, alignedLastBoundary2, alignedLastBar2)

    CycleResult res = CycleResult.new(
        cStart, cEnd,
        sR, eR,
        q1Ts, q2Ts, q3Ts, q4Ts, q5Ts,
        q2ROut, q3ROut, q4ROut, q5ROut,
        qiOut, inW,
        q1, q2, q3, q4, q5
    )

    [stOut, res]

export f_qt_step(QtEngineState eng, int t, int tClose) =>
    QTConfig cfg = eng.cfg
    CycleState st0 = eng.raw

    [st1, dRes] = qt_daily_update(st0, cfg, t, tClose)
    [st2, mRes] = qt_m90_update(st1, cfg, t, tClose)
    [st3, uRes] = qt_micro_update(st2, cfg, t, tClose)
    [st4, nRes] = qt_nano_update(st3, cfg, t, tClose)
    [st5, wRes] = qt_weekly_update(st4, cfg, t, tClose)
    [st6, moRes] = qt_monthly_update(st5, cfg, t, tClose)
    CycleState st7 = CycleState.new(st6.daily, st6.m90, st6.micro, st6.nano, st6.weekly, st6.monthly)

    QtCycleState d = f_qt_cycle_state_from(st0.daily, st1.daily, dRes)
    QtCycleState m = f_qt_cycle_state_from(st0.m90, st2.m90, mRes)
    QtCycleState u = f_qt_cycle_state_from(st0.micro, st3.micro, uRes)
    QtCycleState n = f_qt_cycle_state_from(st0.nano, st4.nano, nRes)
    QtCycleState w = f_qt_cycle_state_from(st4.weekly, st5.weekly, wRes)
    QtCycleState mo = f_qt_cycle_state_from(st5.monthly, st6.monthly, moRes)

    QtEngineState.new(cfg, st7, d, m, u, n, w, mo, eng.cacheVersion, eng.collideVersion)
