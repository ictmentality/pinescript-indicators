f_ssmt_sym_ticker(string sym) =>
    string s = str.upper(sym)
    if str.contains(s, "\"SYMBOL\":\"")
        string[] partsSym = str.split(s, "\"SYMBOL\":\"")
        if array.size(partsSym) > 1
            string right = array.get(partsSym, 1)
            string[] partsEnd = str.split(right, "\"")
            s := array.get(partsEnd, 0)
    if str.contains(s, "{")
        string[] partsBrace = str.split(s, "{")
        s := array.get(partsBrace, 0)
    if str.contains(s, "|")
        string[] partsPipe = str.split(s, "|")
        s := array.get(partsPipe, 0)
    string[] parts = str.split(s, ":")
    array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : s

f_ssmt_match_tag(string sym) =>
    string tk = str.upper(f_ssmt_sym_ticker(sym))
    string tag = ""
    if str.contains(tk, "DXY")
        tag := "DXY"
    else if str.contains(tk, "MNQ") or str.contains(tk, "NQ")
        tag := "NQ"
    else if str.contains(tk, "MES") or str.contains(tk, "ES")
        tag := "ES"
    else if str.contains(tk, "MYM") or str.contains(tk, "YM")
        tag := "YM"
    else if str.contains(tk, "M2K") or str.contains(tk, "RTY")
        tag := "RTY"
    else if str.contains(tk, "MCL") or str.contains(tk, "CL")
        tag := "CL"
    else if str.contains(tk, "MRB") or str.contains(tk, "RB")
        tag := "RB"
    else if str.contains(tk, "MHO") or str.contains(tk, "HO")
        tag := "HO"
    else if str.contains(tk, "M6E") or str.contains(tk, "6E")
        tag := "6E"
    else if str.contains(tk, "M6B") or str.contains(tk, "6B")
        tag := "6B"
    else if str.contains(tk, "ZB")
        tag := "ZB"
    else if str.contains(tk, "TN")
        tag := "TN"
    else if str.contains(tk, "ZF")
        tag := "ZF"

f_check_match(string input_sym) =>
    bool valid = str.length(input_sym) > 0 and str.contains(input_sym, ":")
    if not valid
        false
    else
        string inputUp = str.upper(input_sym)
        string[] parts = str.split(inputUp, ":")
        string tSym = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : inputUp
        string curTicker = str.upper(syminfo.ticker)
        string curId = str.upper(syminfo.tickerid)
        bool directMatch = (curTicker == tSym) or str.contains(curId, tSym) or str.contains(inputUp, curTicker)
        if directMatch
            true
        else
            string inTag = f_ssmt_match_tag(input_sym)
            string curTag = f_ssmt_match_tag(syminfo.tickerid)
            str.length(inTag) > 0 and str.length(curTag) > 0 and inTag == curTag

f_is_same_symbol_exact(string input_sym) =>
    bool valid = str.length(input_sym) > 0 and str.contains(input_sym, ":")
    if not valid
        false
    else
        string inTicker = str.upper(str.trim(str.replace_all(f_ssmt_sym_ticker(input_sym), "=", "")))
        string curTicker = str.upper(str.trim(str.replace_all(f_ssmt_sym_ticker(syminfo.tickerid), "=", "")))
        str.length(inTicker) > 0 and str.length(curTicker) > 0 and inTicker == curTicker

f_ssmt_group_tag(string tag) =>
    int grp = 0
    if tag == "ES" or tag == "NQ" or tag == "YM" or tag == "RTY"
        grp := 1
    else if tag == "CL" or tag == "RB" or tag == "HO"
        grp := 2
    else if tag == "6E" or tag == "6B"
        grp := 3
    else if tag == "GC" or tag == "SI" or tag == "HG"
        grp := 4
    grp

f_ssmt_is_micro_ticker(string tk, string tag) =>
    string m = f_ssmt_micro_prefix(tag)
    str.length(m) > 0 and str.length(tk) >= str.length(m) and str.substring(tk, 0, str.length(m)) == m

f_ssmt_match_chart_contract_size(string inputSym, string chartSym) =>
    string out = inputSym
    string chartT = f_ssmt_sym_ticker(chartSym)
    string chartTag = f_ssmt_match_tag(chartSym)
    int chartGroup = f_ssmt_group_tag(chartTag)
    bool chartMicro = f_ssmt_is_micro_ticker(chartT, chartTag)

    if chartGroup > 0
        string up = str.upper(inputSym)
        string[] parts = str.split(up, ":")
        string ex = array.size(parts) > 1 ? array.get(parts, 0) : ""
        string tk = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : up
        string tkTag = f_ssmt_match_tag(tk)
        int tkGroup = f_ssmt_group_tag(tkTag)
        if tkGroup == chartGroup and str.length(tkTag) > 0
            bool tkMicro = f_ssmt_is_micro_ticker(tk, tkTag)
            if chartMicro != tkMicro
                string oldPrefix = tkMicro ? f_ssmt_micro_prefix(tkTag) : f_ssmt_standard_prefix(tkTag)
                string newPrefix = chartMicro ? f_ssmt_micro_prefix(tkTag) : f_ssmt_standard_prefix(tkTag)
                if str.length(oldPrefix) > 0 and str.length(newPrefix) > 0 and str.length(tk) >= str.length(oldPrefix) and str.substring(tk, 0, str.length(oldPrefix)) == oldPrefix
                    string suffix = str.substring(tk, str.length(oldPrefix), str.length(tk))
                    tk := newPrefix + suffix

        out := str.length(ex) > 0 ? ex + ":" + tk : tk
    out

f_ssmt_apply_futures_modifiers(string sym, bool useBackadjust) =>
    string out = sym
    if syminfo.type == "futures"
        bool isContinuous = str.contains(f_ssmt_sym_ticker(sym), "!")
        if isContinuous
            if useBackadjust
                out := ticker.modify(sym, session=syminfo.session, backadjustment=backadjustment.on)
            else
                out := ticker.modify(sym, session=syminfo.session, backadjustment=backadjustment.off)
        else
            out := ticker.modify(sym, session=syminfo.session)
        string outTrim = str.trim(out)
        if str.length(outTrim) == 0 or outTrim == "="
            out := sym
    out

f_ssmt_chart_dated_suffix(string chartTicker, string chartRoot) =>
    string suffix = ""
    bool hasDated = false
    string ct = str.upper(chartTicker)
    string cr = str.upper(chartRoot)
    int rootLen = str.length(cr)
    bool hasPrefix = rootLen > 0 and str.length(ct) > rootLen and str.substring(ct, 0, rootLen) == cr
    if hasPrefix
        suffix := str.substring(ct, rootLen, str.length(ct))
        string monthCode = str.length(suffix) > 0 ? str.substring(suffix, 0, 1) : ""
        string yearPart = str.length(suffix) > 1 ? str.substring(suffix, 1, str.length(suffix)) : ""
        hasDated := str.contains("FGHJKMNQUVXZ", monthCode) and not na(str.tonumber(yearPart))
    [suffix, hasDated]

f_ssmt_apply_chart_dated_suffix(string inputSym, string chartTicker, string chartRoot) =>
    string out = inputSym
    [chartSuffix, chartHasDated] = f_ssmt_chart_dated_suffix(chartTicker, chartRoot)
    if chartHasDated and str.length(chartSuffix) > 0 and str.length(inputSym) > 0
        string up = str.upper(inputSym)
        string[] parts = str.split(up, ":")
        string ex = array.size(parts) > 1 ? array.get(parts, 0) : ""
        string tk = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : up
        int tkLen = str.length(tk)
        bool isCont = tkLen > 2 and str.substring(tk, tkLen - 2, tkLen) == "1!"
        if isCont
            string root = str.substring(tk, 0, tkLen - 2)
            if str.length(root) > 0
                tk := root + chartSuffix
                out := str.length(ex) > 0 ? ex + ":" + tk : tk
    out

f_ssmt_asset_root(string symTicker, string chartTicker, string chartRoot) =>
    string out = f_ssmt_match_tag(symTicker)
    if str.length(out) == 0
        out := f_ssmt_match_tag(chartTicker)
    if str.length(out) == 0
        out := f_ssmt_match_tag(chartRoot)
    out

export chart_backadjust_debug(bool fallback) =>

    [state, detected] = f_ssmt_backadjust_detect(rawTickerId)

    bool isContinuous = str.contains(chartTk, "!")
    [rawTickerId, chartTk, isContinuous, chartBadj, source, detected]

        st1 := TRIAD1_SYM1

    else if in_triad2

    bool match_1 = f_check_match(st1)
    bool match_2 = f_check_match(st2)

    bool triad_ok = triadModeActive and (in_triad1 or in_triad2 or in_triad3 or in_triad4 or in_triad5 or in_triad6) and (match_1 or match_2 or match_3)

    string tsym3 = ""

            tsym2 := st2
            tsym3 := st3
        else if match_2
            tsym2 := st1
            tsym3 := st3
        else if match_3
            tsym2 := st1
