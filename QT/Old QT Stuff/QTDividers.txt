// Helpers
f_find_int(int[] arr, int v) =>
    int idx = -1
    int i = array.size(arr) - 1
    while i >= 0 and idx == -1
        if array.get(arr, i) == v
            idx := i
        i -= 1
    idx

f_push_unique_int(int[] arr, int v, int maxN) =>
    if f_find_int(arr, v) == -1
        array.push(arr, v)
        while array.size(arr) > maxN
            array.shift(arr)
    0

f_clamp_int(int v, int lo, int hi) =>
    v < lo ? lo : v > hi ? hi : v

f_parent_ts_add(int[] arr, int v) =>
    if not na(v) and f_find_int(arr, v) == -1
        array.push(arr, v)
    0

f_parent_ts_build(int[] outArr, int[] q1Arr, int[] q2Arr, int[] q3Arr, int[] q4Arr, int[] endArr, int cutoff, bool useQ1, bool useQ2, bool useQ3, bool useQ4) =>
    array.clear(outArr)
    int cutoffEff = na(cutoff) ? 0 : cutoff
    int i = array.size(endArr) - 1
    while i >= 0
        int pEnd = array.get(endArr, i)
        if not na(pEnd) and pEnd >= cutoffEff
            if useQ1
                f_parent_ts_add(outArr, array.get(q1Arr, i))
            if useQ2
                f_parent_ts_add(outArr, array.get(q2Arr, i))
            if useQ3
                f_parent_ts_add(outArr, array.get(q3Arr, i))
            if useQ4
                f_parent_ts_add(outArr, array.get(q4Arr, i))
        i -= 1
    0

f_parent_has_ts(int[] tsArr, int[] endArr, int cutoff, int ts) =>
    bool found = false
    int cutoffEff = na(cutoff) ? 0 : cutoff
    if not na(ts)
        int i = array.size(tsArr) - 1
        while i >= 0 and not found
            int pEnd = array.get(endArr, i)
            int pTs = array.get(tsArr, i)
            if not na(pEnd) and pEnd >= cutoffEff and pTs == ts
                found := true
            i -= 1
    found

type vset
    line[] ln
    int[]  ts

vset_new() =>
    vset.new(array.new_line(), array.new_int())

vset_clear(vset s) =>
    int i = array.size(s.ln) - 1
    while i >= 0
        line.delete(array.get(s.ln, i))
        i -= 1
    array.clear(s.ln)
    array.clear(s.ts)
    0

vset_has(vset s, int ts) =>
    bool found = false
    int i = array.size(s.ts) - 1
    while i >= 0 and not found
        found := array.get(s.ts, i) == ts
        i -= 1
    found

vset_add(vset s, int ts, color c) =>
    if not vset_has(s, ts)
        array.push(s.ln, line.new(ts, low, ts, high, xloc=xloc.bar_time, extend=extend.both, style=line.style_solid, width=1, color=f_vcol(c)))
        array.push(s.ts, ts)
    0

vset_trim_since(vset s, int cutoffTs) =>
    int i = array.size(s.ts) - 1
    while i >= 0
        int t = array.get(s.ts, i)
        if t < cutoffTs
            line ln = array.remove(s.ln, i)
            line.delete(ln)
            array.remove(s.ts, i)
        i -= 1
    0

f_set_or_make_vline(line ln, int ts, color c) =>
    line out = ln
    if na(out)
        out := line.new(ts, low, ts, high, xloc=xloc.bar_time, extend=extend.both, style=line.style_solid, width=1, color=f_vcol(c))
    else
        line.set_xy1(out, ts, low)
        line.set_xy2(out, ts, high)
        line.set_style(out, line.style_solid)
        line.set_width(out, 1)
        line.set_color(out, f_vcol(c))
    out

f_set_or_make_hline(line ln, int x1, int x2, float y, color c, sty) =>

        line c = array.get(l3, i)
        line d = array.get(l4, i)

        if not na(b)

        array.set(l1, i, na)
        array.set(l2, i, na)
        array.set(l3, i, na)
        array.set(l4, i, na)
        i -= 1
    0

f_dhist_shift_front() =>
    if array.size(dailyHistStart) > 0
        f_dhist_kill_at(0)
        f_dhist_kill_parent_at(0)
        array.shift(dailyHistStart)
        array.shift(dailyHistEnd)

        array.shift(daily_h_q1o),  array.shift(daily_h_q1oTs)
        array.shift(daily_h_q2o),  array.shift(daily_h_q2oTs)
        array.shift(daily_h_q3o),  array.shift(daily_h_q3oTs)
        array.shift(daily_h_q4o),  array.shift(daily_h_q4oTs)
        array.shift(daily_dfr_hi)
        array.shift(daily_dfr_lo)
        box dfrB = array.shift(daily_dfr_box)
        if not na(dfrB)
            box.delete(dfrB)

        array.shift(daily_l_q1o)
        array.shift(daily_l_q2o)
        array.shift(daily_l_q3o)
        array.shift(daily_l_q4o)

        array.shift(daily_p_l_q1o)
        array.shift(daily_p_l_q2o)
        array.shift(daily_p_l_q3o)
        array.shift(daily_p_l_q4o)
        int dStdvShift = 0
        while dStdvShift < STDV_LEVELS_N and array.size(daily_l_stdv) > 0
            line dStdvLn = array.shift(daily_l_stdv)
            if not na(dStdvLn)
                line.delete(dStdvLn)
            if array.size(daily_l_stdv_lbl) > 0
                label dStdvLbl = array.shift(daily_l_stdv_lbl)
                if not na(dStdvLbl)
                    label.delete(dStdvLbl)
            dStdvShift += 1
    0

f_dhist_trim_cutoff(int cutoffTs) =>
    while array.size(dailyHistStart) > 0 and array.get(dailyHistStart, 0) < cutoffTs
        f_dhist_shift_front()
    0

f_dhist_ensure_current() =>
    if d.inWindow and not na(d.q1Ts)
        int key = d.q1Ts
        int idx = f_find_int(dailyHistStart, key)
        if idx == -1
            array.push(dailyHistStart, key)
            array.push(dailyHistEnd, d.endTs)

            array.push(daily_h_q1o, na), array.push(daily_h_q1oTs, na)
            array.push(daily_h_q2o, na), array.push(daily_h_q2oTs, na)
            array.push(daily_h_q3o, na), array.push(daily_h_q3oTs, na)
            array.push(daily_h_q4o, na), array.push(daily_h_q4oTs, na)
            array.push(daily_dfr_hi, na)
            array.push(daily_dfr_lo, na)
            array.push(daily_dfr_box, na)

            line naL = na
            array.push(daily_l_q1o, naL)
            array.push(daily_l_q2o, naL)
            array.push(daily_l_q3o, naL)
            array.push(daily_l_q4o, naL)
            array.push(daily_p_l_q1o, naL)
            array.push(daily_p_l_q2o, naL)
            array.push(daily_p_l_q3o, naL)
            array.push(daily_p_l_q4o, naL)
            int dStdvPush = 0
            while dStdvPush < STDV_LEVELS_N
                array.push(daily_l_stdv, naL)
                array.push(daily_l_stdv_lbl, na)
                dStdvPush += 1
            idx := array.size(dailyHistStart) - 1

        array.set(dailyHistEnd, idx, d.endTs)

        float q1o = d.q1.o
        int   q1t = d.q1.oTs
        if not na(q1o) and not na(q1t)
            array.set(daily_h_q1o, idx, q1o)
            array.set(daily_h_q1oTs, idx, q1t)

        float q2o = d.q2.o
        int   q2t = d.q2.oTs
        if not na(q2o) and not na(q2t)
            array.set(daily_h_q2o, idx, q2o)
            array.set(daily_h_q2oTs, idx, q2t)

        float q3o = d.q3.o
        int   q3t = d.q3.oTs
        if not na(q3o) and not na(q3t)
            array.set(daily_h_q3o, idx, q3o)
            array.set(daily_h_q3oTs, idx, q3t)

        float q4o = d.q4.o
        int   q4t = d.q4.oTs
        if not na(q4o) and not na(q4t)
            array.set(daily_h_q4o, idx, q4o)
            array.set(daily_h_q4oTs, idx, q4t)

        int dfrStart = na
        if not na(d.q1Ts) and not na(d.q2Ts)
            dfrStart := d.q1Ts + int((d.q2Ts - d.q1Ts) / 3)
        if not na(dfrStart) and time_close > dfrStart and time < d.q2Ts
            float dfrHi = array.get(daily_dfr_hi, idx)
            float dfrLo = array.get(daily_dfr_lo, idx)
            dfrHi := na(dfrHi) or high > dfrHi ? high : dfrHi
            dfrLo := na(dfrLo) or low < dfrLo ? low : dfrLo
            array.set(daily_dfr_hi, idx, dfrHi)
            array.set(daily_dfr_lo, idx, dfrLo)

        while array.size(dailyHistStart) > STORE_MAX
            f_dhist_shift_front()
    0
