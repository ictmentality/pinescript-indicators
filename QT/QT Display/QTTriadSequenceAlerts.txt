f_ssmt_sym_ticker(string sym) =>
    string s = str.upper(sym)
    if str.contains(s, "\"SYMBOL\":\"")
        string[] partsSym = str.split(s, "\"SYMBOL\":\"")
        if array.size(partsSym) > 1
            string right = array.get(partsSym, 1)
            string[] partsEnd = str.split(right, "\"")
            s := array.get(partsEnd, 0)
    if str.contains(s, "{")
        string[] partsBrace = str.split(s, "{")
        s := array.get(partsBrace, 0)
    if str.contains(s, "|")
        string[] partsPipe = str.split(s, "|")
        s := array.get(partsPipe, 0)
    string[] parts = str.split(s, ":")
    array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : s

f_ssmt_match_tag(string sym) =>
    string tk = str.upper(f_ssmt_sym_ticker(sym))
    string tag = ""
    if str.contains(tk, "MNQ") or str.contains(tk, "NQ")
        tag := "NQ"
    else if str.contains(tk, "MES") or str.contains(tk, "ES")
        tag := "ES"
    else if str.contains(tk, "MYM") or str.contains(tk, "YM")
        tag := "YM"
    tag

f_check_match(string input_sym) =>
    bool valid = str.length(input_sym) > 0 and str.contains(input_sym, ":")
    if not valid
        false
    else
        string inputUp = str.upper(input_sym)
        string[] parts = str.split(inputUp, ":")
        string tSym = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : inputUp
        string curTicker = str.upper(syminfo.ticker)
        string curId = str.upper(syminfo.tickerid)
        bool directMatch = (curTicker == tSym) or str.contains(curId, tSym) or str.contains(inputUp, curTicker)
        if directMatch
            true
        else
            string inTag = f_ssmt_match_tag(input_sym)
            string curTag = f_ssmt_match_tag(syminfo.tickerid)
            str.length(inTag) > 0 and str.length(curTag) > 0 and inTag == curTag

f_ssmt_label_sym(string sym) =>
    string allUp = str.upper(f_ssmt_sym_ticker(sym))
    string out = ""
    if str.contains(allUp, "MNQ")
        out := "MNQ"
    else if str.contains(allUp, "MES")
        out := "MES"
    else if str.contains(allUp, "MYM")
        out := "MYM"
    else if str.contains(allUp, "NQ")
        out := "NQ"
    else if str.contains(allUp, "ES")
        out := "ES"
    else if str.contains(allUp, "YM")
        out := "YM"
    if out == ""
        out := allUp
    out

f_candle_state(float o, float c) =>
    int st = 0
    if c > o
        st := 1
    else if c < o
        st := -1
    st

f_tpd_pair_valid(float aC3Open, float aC1Close, float aC2Open, float aC2Close, float bC3Open, float bC1Close, float bC2Open, float bC2Close) =>
    bool ready = not na(aC3Open) and not na(aC1Close) and not na(aC2Open) and not na(aC2Close) and not na(bC3Open) and not na(bC1Close) and not na(bC2Open) and not na(bC2Close)
    int dirA = ready ? (aC3Open > aC1Close ? 1 : aC3Open < aC1Close ? -1 : 0) : 0
    int dirB = ready ? (bC3Open > bC1Close ? 1 : bC3Open < bC1Close ? -1 : 0) : 0
    int c2A = ready ? f_candle_state(aC2Open, aC2Close) : 0
    int c2B = ready ? f_candle_state(bC2Open, bC2Close) : 0
    ready and dirA != 0 and dirB != 0 and dirA != dirB and c2A != c2B

f_bear_cycle_pack(qt.CycleResult cycle, qt.QuarterRecord prev1) =>
    float hCur = float(na)
    int hCurTs = int(na)
    float hPrev = float(na)
    float bhCur = float(na)
    int bhCurTs = int(na)
    float bhPrev = float(na)
    int qCur = int(na)
    if cycle.inWindow and cycle.curQuarterIndex == 1
        hCur := cycle.q1.h
        hCurTs := cycle.q1.hTs
        hPrev := prev1.h
        bhCur := cycle.q1.bh
        bhCurTs := cycle.q1.bhTs
        bhPrev := prev1.bh
        qCur := cycle.q1.boundaryTs
    else if cycle.inWindow and cycle.curQuarterIndex == 2
        hCur := cycle.q2.h
        hCurTs := cycle.q2.hTs
        hPrev := cycle.q1.h
        bhCur := cycle.q2.bh
        bhCurTs := cycle.q2.bhTs
        bhPrev := cycle.q1.bh
        qCur := cycle.q2.boundaryTs
    else if cycle.inWindow and cycle.curQuarterIndex == 3
        hCur := cycle.q3.h
        hCurTs := cycle.q3.hTs
        hPrev := cycle.q2.h
        bhCur := cycle.q3.bh
        bhCurTs := cycle.q3.bhTs
        bhPrev := cycle.q2.bh
        qCur := cycle.q3.boundaryTs
    else if cycle.inWindow and cycle.curQuarterIndex == 4
        hCur := cycle.q4.h
        hCurTs := cycle.q4.hTs
        hPrev := cycle.q3.h
        bhCur := cycle.q4.bh
        bhCurTs := cycle.q4.bhTs
        bhPrev := cycle.q3.bh
        qCur := cycle.q4.boundaryTs
    else if cycle.inWindow and cycle.curQuarterIndex == 5
        hCur := cycle.q5.h
        hCurTs := cycle.q5.hTs
        hPrev := cycle.q4.h
        bhCur := cycle.q5.bh
        bhCurTs := cycle.q5.bhTs
        bhPrev := cycle.q4.bh
        qCur := cycle.q5.boundaryTs
    [hCur, hCurTs, hPrev, bhCur, bhCurTs, bhPrev, qCur]

f_weekly_bear_pack_expr() =>
    var qt.QTConfig peerCfg = qt.qt_config_default()
    var qt.CycleState peerSt = qt.qt_state_new()
    var int seenCycleStart = int(na)
    var float currentCycleTwo = float(na)
    var float previousCycleTwo = float(na)

    [peerStNext, peerWk] = qt.qt_weekly_update(peerSt, peerCfg, time, time_close)
    peerSt := peerStNext
    [_, _, _, _, _, pwprev1, _] = qt.qt_weekly_qr_debug(peerSt)

    if not na(peerWk.startTs) and (na(seenCycleStart) or peerWk.startTs != seenCycleStart)
        if not na(currentCycleTwo)
            previousCycleTwo := currentCycleTwo
        currentCycleTwo := float(na)
        seenCycleStart := peerWk.startTs
    if not na(peerWk.q2.o)
        currentCycleTwo := peerWk.q2.o

    float activeTwo = peerWk.curQuarterIndex == 1 ? previousCycleTwo : currentCycleTwo
    [hCur, hCurTs, hPrev, bhCur, bhCurTs, bhPrev, qCur] = f_bear_cycle_pack(peerWk, pwprev1)
    [hCur, hCurTs, hPrev, bhCur, bhCurTs, bhPrev, qCur, peerWk.startTs, peerWk.curQuarterIndex, activeTwo]

f_daily_bear_pack_expr() =>
    var qt.QTConfig peerCfg = qt.qt_config_default()
    var qt.CycleState peerSt = qt.qt_state_new()

    [peerStNext, peerDay] = qt.qt_daily_update(peerSt, peerCfg, time, time_close)
    peerSt := peerStNext
    [_, _, _, _, _, pdprev1, _] = qt.qt_daily_qr_debug(peerSt)
    [hCur, hCurTs, hPrev, bhCur, bhCurTs, bhPrev, qCur] = f_bear_cycle_pack(peerDay, pdprev1)
    [hCur, hCurTs, hPrev, bhCur, bhCurTs, bhPrev, qCur]

f_h4_tpd_pack_expr() =>
    [open, close[2], open[1], close[1], time, time_close]

f_bear_ssmt_pair(bool hasA, float aHCur, int aHCurTs, float aHPrev, bool hasB, float bHCur, int bHCurTs, float bHPrev) =>
    bool sweepA = hasA ? (aHCur > aHPrev) : false
    bool sweepB = hasB ? (bHCur > bHPrev) : false
    bool split = hasA and hasB and sweepA != sweepB
    int eventTs = split ? (sweepA ? aHCurTs : bHCurTs) : int(na)
    [split, eventTs]

f_pick_event(bool isValid, int eventTs, string pairName, int pairCode, int eventType, bool currentValid, int currentTs, string currentPair, int currentPairCode, int currentType) =>
    bool outValid = currentValid
    int outTs = currentTs
    string outPair = currentPair
    int outCode = currentPairCode
    int outType = currentType
    if isValid and not na(eventTs) and (not outValid or na(outTs) or eventTs > outTs)
        outValid := true
        outTs := eventTs
        outPair := pairName
        outCode := pairCode
        outType := eventType
    [outValid, outTs, outPair, outCode, outType]
