// ============================================================================
// HELPERS
// ============================================================================
f_kill(line ln) =>
    line out = na
    if not na(ln)
        line.delete(ln)
    out

f_set_or_make_vline(line ln, int ts, color c, int w) =>
    line out = ln
    if na(ts)
        out := f_kill(out)
    else
        if na(out)
            out := line.new(ts, low, ts, high, xloc=xloc.bar_time, extend=extend.both, style=line.style_solid, width=w, color=c)
        else
            line.set_xy1(out, ts, low)
            line.set_xy2(out, ts, high)
            line.set_extend(out, extend.both)
            line.set_style(out, line.style_solid)
            line.set_width(out, w)
            line.set_color(out, c)
    out

f_ts_for_draw(bool enabled, int ts) =>
    enabled and not na(ts) and (showProjected or time_close > ts) ? ts : int(na)

f_draw_cycle(
    bool enabled,
    int q1Ts,
    int q2Ts,
    int q3Ts,
    int q4Ts,
    int q5Ts,
    int endTs,
    line l1,
    line l2,
    line l3,
    line l4,
    line l5,
    line lE
) =>
    int dQ1 = f_ts_for_draw(enabled, q1Ts)
    int dQ2 = f_ts_for_draw(enabled, q2Ts)
    int dQ3 = f_ts_for_draw(enabled, q3Ts)
    int dQ4 = f_ts_for_draw(enabled, q4Ts)
    int dQ5 = f_ts_for_draw(enabled, q5Ts)
    int dE  = f_ts_for_draw(enabled, endTs)

    line o1 = f_set_or_make_vline(l1, dQ1, COL_Q1, lineWidth)
    line o2 = f_set_or_make_vline(l2, dQ2, COL_Q2, lineWidth)
    line o3 = f_set_or_make_vline(l3, dQ3, COL_Q3, lineWidth)
    line o4 = f_set_or_make_vline(l4, dQ4, COL_Q4, lineWidth)
    line o5 = f_set_or_make_vline(l5, dQ5, COL_Q5, lineWidth)
    line oE = f_set_or_make_vline(lE, dE, COL_Q1, lineWidth)

    line[] out = array.new_line()
    array.push(out, o1)
    array.push(out, o2)
    array.push(out, o3)
    array.push(out, o4)
    array.push(out, o5)
    array.push(out, oE)
    out
