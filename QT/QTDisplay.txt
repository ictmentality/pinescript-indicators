f_qt_day_start_raw(int t, string tz, int dayStartHour, int dayStartMin) =>
    int y  = year(t, tz)
    int mo = month(t, tz)
    int da = dayofmonth(t, tz)
    int h  = hour(t, tz)
    int mi = minute(t, tz)
    int base = timestamp(tz, y, mo, da, dayStartHour, dayStartMin)
    int out  = (h > dayStartHour or (h == dayStartHour and mi >= dayStartMin)) ? base : base - LEN_24H
    out

f_qt_day_start(int t) =>
    f_qt_day_start_raw(t, TZ, 18, 0)

f_next_ds_after(int t) =>
    int ds0 = f_qt_day_start(t)
    ds0 <= t ? ds0 + LEN_24H : ds0

f_ceil_to_1h(int ts) =>
    int step = 60 * MS_MIN
    int rem  = ts % step
    rem == 0 ? ts : ts + (step - rem)

f_weekly_aligned_open_expr(int q1Ts, int q2Ts, int q3Ts, int q4Ts, int q5Ts, int endTs) =>
    int tRef = time
    bool inW = tRef >= q1Ts and tRef < endTs

    int qi = 1
    if inW
        if tRef >= q2Ts
            qi := 2
        if tRef >= q3Ts
            qi := 3
        if tRef >= q4Ts
            qi := 4
        if tRef >= q5Ts
            qi := 5

    int qCurTs = int(na)
    if inW
        qCurTs := qi == 1 ? q1Ts : qi == 2 ? q2Ts : qi == 3 ? q3Ts : qi == 4 ? q4Ts : q5Ts

    int targetTs = na(qCurTs) ? int(na) : f_ceil_to_1h(qCurTs)

    var int   lastTarget = int(na)
    var float actualOpen = float(na)
    var int   actualOpenTs = int(na)
    var int   actualBoundaryTs = int(na)
    var int   actualQi = int(na)

    if na(lastTarget) or targetTs != lastTarget
        lastTarget := targetTs
        actualOpen := float(na)
        actualOpenTs := int(na)
        actualBoundaryTs := int(na)
        actualQi := int(na)

    bool hit = not na(targetTs) and (time >= targetTs) and (na(time[1]) or time[1] < targetTs)
    if hit and na(actualOpen)
        actualOpen := open
        actualOpenTs := time
        actualBoundaryTs := qCurTs
        actualQi := qi

    [actualOpen, actualOpenTs, actualBoundaryTs, actualQi]

f_ssmt_sym_ticker(string sym) =>
    string s = str.upper(sym)
    if str.contains(s, "{")
        string[] partsBrace = str.split(s, "{")
        s := array.get(partsBrace, 0)
    if str.contains(s, "|")
        string[] partsPipe = str.split(s, "|")
        s := array.get(partsPipe, 0)
    string[] parts = str.split(s, ":")
    array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : s

f_ssmt_norm_sym(string sym) =>
    syminfo.type == "futures" ? ticker.modify(sym, session=syminfo.session) : sym

f_check_match(string input_sym) =>
    bool valid = str.length(input_sym) > 0 and str.contains(input_sym, ":")
    if not valid
        false
    else
        string inputUp = str.upper(input_sym)
        string[] parts = str.split(inputUp, ":")
        string tSym = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : inputUp
        string curTicker = str.upper(syminfo.ticker)
        string curId = str.upper(syminfo.tickerid)
        (curTicker == tSym) or str.contains(curId, tSym) or str.contains(inputUp, curTicker)

f_ssmt_tag(string sym) =>
    string s = f_ssmt_sym_ticker(sym)
    string tag = s
    if str.contains(s, "MES")
        tag := "ES"
    else if str.contains(s, "MNQ")
        tag := "NQ"
    else if str.contains(s, "MYM")
        tag := "YM"
    else if str.contains(s, "ES")
        tag := "ES"
    else if str.contains(s, "NQ")
        tag := "NQ"
    else if str.contains(s, "YM")
        tag := "YM"
    tag

        out := "PL"

        out := "HG"
    else if str.contains(allUp, "CL")
        out := "CL"

    string s = f_ssmt_sym_ticker(sym)
    s := str.replace_all(s, "}", "")
    s := str.replace_all(s, "{", "")
    s := str.replace_all(s, "\"", "")
    s := str.replace_all(s, "'", "")
    s := str.replace_all(s, "=", "")
    s := str.trim(s)
    string up = str.upper(s)
    if out == ""
        out := up
        if str.length(up) >= 3
            string r3 = str.substring(up, 0, 3)
            if r3 == "MES" or r3 == "MNQ" or r3 == "MYM" or r3 == "MGC" or r3 == "SIL" or r3 == "MHG"
                out := r3
        if out == up and str.length(up) >= 2
            string r2 = str.substring(up, 0, 2)
            if r2 == "ES" or r2 == "NQ" or r2 == "YM" or r2 == "GC" or r2 == "SI" or r2 == "PL" or r2 == "HG" or r2 == "CL"
                out := r2
    out

f_ssmt_match_chart_contract_size(string inputSym, string chartSym) =>
    string out = inputSym
    string chartT = f_ssmt_sym_ticker(chartSym)

    bool chartMicro = str.contains(chartT, "MES") or str.contains(chartT, "MNQ") or str.contains(chartT, "MYM")
    bool chartMini = (str.contains(chartT, "ES") and not str.contains(chartT, "MES")) or (str.contains(chartT, "NQ") and not str.contains(chartT, "MNQ")) or (str.contains(chartT, "YM") and not str.contains(chartT, "MYM"))

    if chartMicro or chartMini
        string up = str.upper(inputSym)
        string[] parts = str.split(up, ":")
        string ex = array.size(parts) > 1 ? array.get(parts, 0) : ""
        string tk = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : up

        bool tkMicro = str.length(tk) >= 3 and (str.substring(tk, 0, 3) == "MES" or str.substring(tk, 0, 3) == "MNQ" or str.substring(tk, 0, 3) == "MYM")
        bool tkMini = str.length(tk) >= 2 and (str.substring(tk, 0, 2) == "ES" or str.substring(tk, 0, 2) == "NQ" or str.substring(tk, 0, 2) == "YM")

        if chartMicro and tkMini
            string root2 = str.substring(tk, 0, 2)
            string suffix2 = str.substring(tk, 2, str.length(tk))
            if root2 == "ES"
                tk := "MES" + suffix2
            else if root2 == "NQ"
                tk := "MNQ" + suffix2
            else if root2 == "YM"
                tk := "MYM" + suffix2
        else if chartMini and tkMicro
            string root3 = str.substring(tk, 0, 3)
            string suffix3 = str.substring(tk, 3, str.length(tk))
            if root3 == "MES"
                tk := "ES" + suffix3
            else if root3 == "MNQ"
                tk := "NQ" + suffix3
            else if root3 == "MYM"
                tk := "YM" + suffix3

        out := str.length(ex) > 0 ? ex + ":" + tk : tk
    out

f_next_ds_after_cfg(int t, WeeklyConfig cfg) =>

f_week_n(int[] startsIn, int[] endsIn, float[] woIn, float[] twoIn) =>
    int nS = array.size(startsIn)
    int nE = array.size(endsIn)
    int nW = array.size(woIn)
    int nT = array.size(twoIn)
    int n0 = nS < nE ? nS : nE
    int n1 = nW < nT ? nW : nT
    n0 < n1 ? n0 : n1

f_weekly_scan_update(
    WeeklyState stIn,
    WeeklyConfig cfg,
    int t,
    int tClose,
    int wkStartTs,
    int wkEndProjTs,
    float wkOpen,
    bool enabled
) =>
    int[] starts2 = array.copy(stIn.starts)
    int[] ends2 = array.copy(stIn.ends)
    float[] wo2 = array.copy(stIn.wo)
    float[] two2 = array.copy(stIn.two)
    float[] three2 = array.copy(stIn.three)
    float[] four2 = array.copy(stIn.four)
    float[] five2 = array.copy(stIn.five)
    int[] woTs2 = array.copy(stIn.woTs)
    int[] twoTs2 = array.copy(stIn.twoTs)
    int[] threeTs2 = array.copy(stIn.threeTs)
    int[] fourTs2 = array.copy(stIn.fourTs)
    int[] fiveTs2 = array.copy(stIn.fiveTs)
    int lastStart2 = stIn.lastStartTs

    WeeklyResult res = f_weekly_result_empty()

    if enabled and not na(wkStartTs)
        bool cycleChanged = na(lastStart2) or wkStartTs != lastStart2

        if cycleChanged
            int nEnds = array.size(ends2)
            if bar_index > 0 and nEnds > 0
                array.set(ends2, nEnds - 1, time_close[1])

            array.push(starts2, wkStartTs)
            array.push(ends2, int(na))
            array.push(wo2, wkOpen)
            array.push(two2, float(na))
            array.push(three2, float(na))
            array.push(four2, float(na))
            array.push(five2, float(na))
            array.push(woTs2, wkStartTs)
            array.push(twoTs2, int(na))
            array.push(threeTs2, int(na))
            array.push(fourTs2, int(na))
            array.push(fiveTs2, int(na))
            lastStart2 := wkStartTs

            while array.size(starts2) > cfg.keepCycles
                array.shift(starts2)
                array.shift(ends2)
                array.shift(wo2)
                array.shift(two2)
                array.shift(three2)
                array.shift(four2)
                array.shift(five2)
                array.shift(woTs2)
                array.shift(twoTs2)
                array.shift(threeTs2)
                array.shift(fourTs2)
                array.shift(fiveTs2)
        else
            int nWo = array.size(wo2)
            if nWo > 0
                array.set(wo2, nWo - 1, wkOpen)
                int woTsCur = array.get(woTs2, nWo - 1)
                if na(woTsCur)
                    array.set(woTs2, nWo - 1, wkStartTs)

        int n = f_week_n(starts2, ends2, wo2, two2)
        if n > 0
            int curWs = array.get(starts2, n - 1)
            int curQ2 = f_next_ds_after_cfg(curWs, cfg)

            int curWeStored = array.get(ends2, n - 1)
            int curWe = na(curWeStored) ? wkEndProjTs : curWeStored
            int curQ3 = curQ2 + LEN_24H
            int curQ4 = curQ3 + LEN_24H
            int curQ5 = curQ4 + LEN_24H

            [aOpen, aOpenTs, aBoundaryTs, aQi] = f_weekly_aligned_open_current(curWs, curQ2, curQ3, curQ4, curQ5, curWe)
            if not na(aOpen) and not na(aBoundaryTs)
                if aBoundaryTs == curWs
                    array.set(wo2, n - 1, aOpen)
                    array.set(woTs2, n - 1, na(aOpenTs) ? curWs : aOpenTs)
                else if aBoundaryTs == curQ2
                    float curTwoOpen = array.get(two2, n - 1)
                    if na(curTwoOpen)
                        array.set(two2, n - 1, aOpen)
                        array.set(twoTs2, n - 1, na(aOpenTs) ? curQ2 : aOpenTs)
                else if aBoundaryTs == curQ3
                    float curThree = array.get(three2, n - 1)
                    if na(curThree)
                        array.set(three2, n - 1, aOpen)
                        array.set(threeTs2, n - 1, na(aOpenTs) ? curQ3 : aOpenTs)
                else if aBoundaryTs == curQ4
                    float curFour = array.get(four2, n - 1)
                    if na(curFour)
                        array.set(four2, n - 1, aOpen)
                        array.set(fourTs2, n - 1, na(aOpenTs) ? curQ4 : aOpenTs)
                else if aBoundaryTs == curQ5
                    float curFive = array.get(five2, n - 1)
                    if na(curFive)
                        array.set(five2, n - 1, aOpen)
                        array.set(fiveTs2, n - 1, na(aOpenTs) ? curQ5 : aOpenTs)

            float q1Open = array.get(wo2, n - 1)
            float q2Open = array.get(two2, n - 1)
            float q3Open = array.get(three2, n - 1)
            float q4Open = array.get(four2, n - 1)
            float q5Open = array.get(five2, n - 1)
            int q1OpenTs = array.get(woTs2, n - 1)
            int q2OpenTs = array.get(twoTs2, n - 1)
            int q3OpenTs = array.get(threeTs2, n - 1)
            int q4OpenTs = array.get(fourTs2, n - 1)
            int q5OpenTs = array.get(fiveTs2, n - 1)

            res := WeeklyResult.new(
                curWs,
                curWe,
                tClose > curWs,
                tClose > curWe,
                curWs,
                curQ2,
                curQ3,
                curQ4,
                curQ5,
                tClose > curQ2,
                tClose > curQ3,
                tClose > curQ4,
                tClose > curQ5,
                q1Open,
                q2Open,
                q3Open,
                q4Open,
                q5Open,
                q1OpenTs,
                q2OpenTs,
                q3OpenTs,
                q4OpenTs,
                q5OpenTs
            )

    WeeklyState stOut = WeeklyState.new(starts2, ends2, wo2, two2, three2, four2, five2, woTs2, twoTs2, threeTs2, fourTs2, fiveTs2, lastStart2)
    [stOut, res]

type vset
    line[] ln
    int[]  ts

vset_new() =>
    vset.new(array.new_line(), array.new_int())

vset_clear(vset s) =>
    int i = s.ln.size() - 1
    while i >= 0
        line.delete(s.ln.get(i))
        i -= 1
    s.ln.clear()
    s.ts.clear()
    0

vset_trim_since(vset s, int cutoff) =>
    int i = s.ts.size() - 1
    while i >= 0
        if s.ts.get(i) < cutoff
            line.delete(s.ln.get(i))
            array.remove(s.ln, i)
            array.remove(s.ts, i)
        i -= 1
    0

vset_has(vset s, int ts) =>
    bool found = false
    int i = s.ts.size() - 1
    while i >= 0 and not found
        found := s.ts.get(i) == ts
        i -= 1
    found

vset_add(vset s, int ts, color c) =>
    if not vset_has(s, ts)
        s.ln.push(line.new(ts, low, ts, high, xloc=xloc.bar_time, extend=extend.both, style=line.style_solid, width=1, color=f_vcol(c)))
        s.ts.push(ts)
    0

type hset
    line[] ln
    int[]  st

hset_new() =>
    hset.new(array.new_line(), array.new_int())

hset_clear(hset s) =>
    int i = s.ln.size() - 1
    while i >= 0
        line.delete(s.ln.get(i))
        i -= 1
    s.ln.clear()
    s.st.clear()
    0

hset_trim_since(hset s, int cutoff) =>
    int i = s.st.size() - 1
    while i >= 0
        if s.st.get(i) < cutoff
            line.delete(s.ln.get(i))
            array.remove(s.ln, i)
            array.remove(s.st, i)
        i -= 1
    0

hset_has(hset s, int st) =>
    bool found = false
    int i = s.st.size() - 1
    while i >= 0 and not found
        found := s.st.get(i) == st
        i -= 1
    found

hset_add(hset s, int st, int et, float y, color c, string styleStr) =>
    if not hset_has(s, st)
        s.ln.push(line.new(st, y, et, y, xloc=xloc.bar_time, extend=extend.none, style=lineStyleFrom(styleStr), width=1, color=c))
        s.st.push(st)
    0

hset_sync(hset s, color c, string styleStr) =>
    int i = s.ln.size() - 1
    while i >= 0
        line ln = s.ln.get(i)
        if not na(ln)
            line.set_color(ln, c)
            line.set_style(ln, lineStyleFrom(styleStr))
            line.set_width(ln, 1)
        i -= 1
    0

f_ssmt_hist_push(line ln, label lb, line[] lines, label[] labels, int maxN) =>
    if not na(ln) or not na(lb)
        if maxN <= 0
            if not na(ln)
                line.delete(ln)
            if not na(lb)
                label.delete(lb)
        else
            array.push(lines, ln)
            array.push(labels, lb)
            while array.size(lines) > maxN
                line oldLn = array.shift(lines)
                if not na(oldLn)
                    line.delete(oldLn)
                label oldLb = array.shift(labels)
                if not na(oldLb)
                    label.delete(oldLb)
    0

f_ssmt_hist_trim(line[] lines, label[] labels, int maxN) =>
    while array.size(lines) > maxN
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    while array.size(labels) > maxN
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

f_ssmt_hist_clear(line[] lines, label[] labels) =>
    while array.size(lines) > 0
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
    while array.size(labels) > 0
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

f_kill(line ln) =>
    line out = na
    if not na(ln)
        line.delete(ln)
    out

f_set_or_kill_open_line(line ln, bool shouldShow, int st, int et, float y, color c, string styleStr) =>
    line out = ln
    bool valid = shouldShow and not na(st) and not na(et) and et > st and not na(y)
    if valid
        out := f_set_or_make_hline_w_style(out, st, et, y, c, 1, styleStr)
    else
        out := f_kill(out)
    out

f_week_inputs_from_engine(int[] startsIn, int curEnd, float curQ1Open, float curQ2Open) =>
    int[] starts = array.copy(startsIn)
    int[] ends = array.new_int()
    float[] wo = array.new_float()
    float[] two = array.new_float()

    int n = array.size(starts)
    if n > 0
        int i = 0
        while i < n
            int endTs = i < (n - 1) ? array.get(starts, i + 1) : curEnd
            array.push(ends, endTs)
            array.push(wo, float(na))
            array.push(two, float(na))
            i += 1
        array.set(wo, n - 1, curQ1Open)
        array.set(two, n - 1, curQ2Open)

    [starts, ends, wo, two]

// ============================================================================
// FIXED CYCLE RUNNER (Daily / m90 / Micro / Nano)
// ============================================================================
run_fixed_cycle(
    int[]   startsIn,
    int[]   endsIn,
    float[] woIn,
    float[] twoIn,
    int     endProj,
    int     curQ2Ts,
    int     curQ3Ts,
    int     curQ4Ts,
    bool    curQ2Realized,
    bool    curQ3Realized,
    bool    curQ4Realized,
    float   curQ3Open,
    float   curQ4Open,
    int     keepN,
    vset    v,
    hset    hWO,
    hset    hTWO,
    hset    hQ3,
    hset    hQ4,
    line    projQ2,
    line    projQ3,
    line    projQ4,
    line    projEnd
) =>
    line p2 = projQ2
    line p3 = projQ3
    line p4 = projQ4
    line pE = projEnd

    int nS = startsIn.size()
    int nE = endsIn.size()
    int nW = woIn.size()
    int nT = twoIn.size()
    int n0 = nS < nE ? nS : nE
    int n1 = nW < nT ? nW : nT
    int n  = n0 < n1 ? n0 : n1

    if n <= 0
        p2 := f_kill(p2)
        p3 := f_kill(p3)
        p4 := f_kill(p4)
        pE := f_kill(pE)
        line[] out0 = array.new_line()
        array.push(out0, p2)
        array.push(out0, p3)
        array.push(out0, p4)
        array.push(out0, pE)
        out0
    else
        int first = math.max(0, n - keepN)
        int cut   = startsIn.get(first)

        vset_trim_since(v, cut)
        if showOpens
            hset_trim_since(hWO, cut)
            hset_trim_since(hTWO, cut)
            hset_trim_since(hQ3, cut)
            hset_trim_since(hQ4, cut)

        int i = first
        while i < n
            int s = startsIn.get(i)
            int eStored = endsIn.get(i)
            int e = na(eStored) ? endProj : eStored
            bool isCur = i == (n - 1)

            if s >= cut and time_close > s
                vset_add(v, s, COL_Q1)

            if isCur
                if curQ2Realized and not na(curQ2Ts) and curQ2Ts >= cut and curQ2Ts < e
                    vset_add(v, curQ2Ts, COL_Q2)
                if curQ3Realized and not na(curQ3Ts) and curQ3Ts >= cut and curQ3Ts < e
                    vset_add(v, curQ3Ts, COL_Q3)
                if curQ4Realized and not na(curQ4Ts) and curQ4Ts >= cut and curQ4Ts < e
                    vset_add(v, curQ4Ts, COL_Q4)

            if not na(eStored) and eStored >= cut and time_close > eStored
                vset_add(v, eStored, COL_Q1)

            if showOpens
                float wo = woIn.get(i)
                if not na(wo) and time_close > s
                    hset_add(hWO, s, e, wo, f_vcol(COL_Q1), q1OpenStyleStr)

                float twoO = twoIn.get(i)
                if isCur and not na(twoO) and not na(curQ2Ts) and time_close > curQ2Ts
                    hset_add(hTWO, curQ2Ts, e, twoO, f_vcol(COL_Q2), trueOpenStyleStr)

                if isCur and curQ3Realized and not na(curQ3Open) and not na(curQ3Ts) and not na(curQ4Ts) and curQ4Ts > curQ3Ts and time_close > curQ3Ts
                    hset_add(hQ3, curQ3Ts, curQ4Ts, curQ3Open, f_vcol(COL_Q3), trueOpenStyleStr)

                if isCur and curQ4Realized and not na(curQ4Open) and not na(curQ4Ts) and e > curQ4Ts and time_close > curQ4Ts
                    hset_add(hQ4, curQ4Ts, e, curQ4Open, f_vcol(COL_Q4), trueOpenStyleStr)

            i += 1

        if showOpens
            hset_sync(hWO, f_vcol(COL_Q1), q1OpenStyleStr)
            hset_sync(hTWO, f_vcol(COL_Q2), trueOpenStyleStr)
            hset_sync(hQ3, f_vcol(COL_Q3), trueOpenStyleStr)
            hset_sync(hQ4, f_vcol(COL_Q4), trueOpenStyleStr)

        if not na(curQ2Ts) and not curQ2Realized and time_close <= curQ2Ts
            p2 := f_set_or_make_vline(p2, curQ2Ts, COL_Q2)
        else
            p2 := f_kill(p2)

        if not na(curQ3Ts) and not curQ3Realized and time_close <= curQ3Ts
            p3 := f_set_or_make_vline(p3, curQ3Ts, COL_Q3)
        else
            p3 := f_kill(p3)

        if not na(curQ4Ts) and not curQ4Realized and time_close <= curQ4Ts
            p4 := f_set_or_make_vline(p4, curQ4Ts, COL_Q4)
        else
            p4 := f_kill(p4)

        pE := f_set_or_make_vline(pE, endProj, COL_Q1)

        line[] out = array.new_line()
        array.push(out, p2)
        array.push(out, p3)
        array.push(out, p4)
        array.push(out, pE)
        out

// ============================================================================
// WEEKLY CYCLE RUNNER
// ============================================================================
run_week_cycle(
    int[]   wkStarts,
    int[]   wkEnds,
    float[] wkWo,
    float[] wkTwo,
    int     wEndProj,
    int     curQ2Ts,
    int     curQ3Ts,
    int     curQ4Ts,
    int     curQ5Ts,
    bool    curQ2Realized,
    bool    curQ3Realized,
    bool    curQ4Realized,
    bool    curQ5Realized,
    float   curQ3Open,
    float   curQ4Open,
    float   curQ5Open,
    int     keepWeeks,
    vset    v,
    hset    hWO,
    hset    hTWO,
    hset    hQ3,
    hset    hQ4,
    hset    hQ5,
    line    projQ2,
    line    projQ3,
    line    projQ4,
    line    projQ5,
    line    projEnd
) =>
    line p2  = projQ2
    line p3  = projQ3
    line p4  = projQ4
    line p5  = projQ5
    line pE  = projEnd

    int nS = wkStarts.size()
    int nE = wkEnds.size()
    int nW = wkWo.size()
    int nT = wkTwo.size()
    int n0 = nS < nE ? nS : nE
    int n1 = nW < nT ? nW : nT
    int n  = n0 < n1 ? n0 : n1

    if n <= 0
        p2 := f_kill(p2)
        p3 := f_kill(p3)
        p4 := f_kill(p4)
        p5 := f_kill(p5)
        pE := f_kill(pE)
        line[] out0 = array.new_line()
        array.push(out0, p2)
        array.push(out0, p3)
        array.push(out0, p4)
        array.push(out0, p5)
        array.push(out0, pE)
        out0
    else
        int first = math.max(0, n - keepWeeks)
        int cut   = wkStarts.get(first)

        vset_trim_since(v, cut)
        if showOpens
            hset_trim_since(hWO, cut)
            hset_trim_since(hTWO, cut)
            hset_trim_since(hQ3, cut)
            hset_trim_since(hQ4, cut)
            hset_trim_since(hQ5, cut)

        int i = first
        while i < n
            int wS = wkStarts.get(i)
            int wEStored = wkEnds.get(i)
            int wE = na(wEStored) ? wEndProj : wEStored
            bool isCurWeek = i == (n - 1)

            if wS >= cut and time_close > wS
                vset_add(v, wS, COL_Q1)

            if isCurWeek
                if curQ2Realized and not na(curQ2Ts) and curQ2Ts >= cut and curQ2Ts < wE
                    vset_add(v, curQ2Ts, COL_Q2)
                if curQ3Realized and not na(curQ3Ts) and curQ3Ts >= cut and curQ3Ts < wE
                    vset_add(v, curQ3Ts, COL_Q3)
                if curQ4Realized and not na(curQ4Ts) and curQ4Ts >= cut and curQ4Ts < wE
                    vset_add(v, curQ4Ts, COL_Q4)
                if curQ5Realized and not na(curQ5Ts) and curQ5Ts >= cut and curQ5Ts < wE
                    vset_add(v, curQ5Ts, COL_Q5)

            if not na(wEStored) and wEStored >= cut and time_close > wEStored
                vset_add(v, wEStored, COL_Q1)

            if showOpens
                float wo = wkWo.get(i)
                if not na(wo) and time_close > wS
                    hset_add(hWO, wS, wE, wo, f_vcol(COL_Q1), q1OpenStyleStr)

                float twoO = wkTwo.get(i)
                if isCurWeek and not na(twoO) and not na(curQ2Ts) and time_close > curQ2Ts
                    hset_add(hTWO, curQ2Ts, wE, twoO, f_vcol(COL_Q2), trueOpenStyleStr)

                if isCurWeek and curQ3Realized and not na(curQ3Open) and not na(curQ3Ts) and not na(curQ4Ts) and curQ4Ts > curQ3Ts and time_close > curQ3Ts
                    hset_add(hQ3, curQ3Ts, curQ4Ts, curQ3Open, f_vcol(COL_Q3), trueOpenStyleStr)

                int q4End = not na(curQ5Ts) and not na(curQ4Ts) and curQ5Ts > curQ4Ts ? curQ5Ts : wE
                if isCurWeek and curQ4Realized and not na(curQ4Open) and not na(curQ4Ts) and q4End > curQ4Ts and time_close > curQ4Ts
