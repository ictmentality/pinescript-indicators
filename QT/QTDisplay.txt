f_ssmt_sym_ticker(string sym) =>
    string s = str.upper(sym)
    if str.contains(s, "{")
        string[] partsBrace = str.split(s, "{")
        s := array.get(partsBrace, 0)
    if str.contains(s, "|")
        string[] partsPipe = str.split(s, "|")
        s := array.get(partsPipe, 0)
    string[] parts = str.split(s, ":")
    array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : s

f_ssmt_norm_sym(string sym) =>
    syminfo.type == "futures" ? ticker.modify(sym, session=syminfo.session) : sym

f_check_match(string input_sym) =>
    bool valid = str.length(input_sym) > 0 and str.contains(input_sym, ":")
    if not valid
        false
    else
        string inputUp = str.upper(input_sym)
        string[] parts = str.split(inputUp, ":")
        string tSym = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : inputUp
        string curTicker = str.upper(syminfo.ticker)
        string curId = str.upper(syminfo.tickerid)
        (curTicker == tSym) or str.contains(curId, tSym) or str.contains(inputUp, curTicker)

f_is_same_symbol_exact(string input_sym) =>
    bool valid = str.length(input_sym) > 0 and str.contains(input_sym, ":")
    if not valid
        false
    else
        string inTicker = str.upper(str.trim(str.replace_all(f_ssmt_sym_ticker(input_sym), "=", "")))
        string curTicker = str.upper(str.trim(str.replace_all(f_ssmt_sym_ticker(syminfo.tickerid), "=", "")))
        str.length(inTicker) > 0 and str.length(curTicker) > 0 and inTicker == curTicker

f_ssmt_tag(string sym) =>
    string s = f_ssmt_sym_ticker(sym)
    string tag = s
    if str.contains(s, "MES")
        tag := "ES"
    else if str.contains(s, "MNQ")
        tag := "NQ"
    else if str.contains(s, "MYM")
        tag := "YM"
    else if str.contains(s, "ES")
        tag := "ES"
    else if str.contains(s, "NQ")
        tag := "NQ"
    else if str.contains(s, "YM")
        tag := "YM"
    tag

f_ssmt_label_sym(string sym) =>
    string allUp = str.upper(f_ssmt_sym_ticker(sym))
    string out = ""
    if str.contains(allUp, "MNQ")
        out := "MNQ"
    else if str.contains(allUp, "MES")
        out := "MES"
    else if str.contains(allUp, "MYM")
        out := "MYM"
    else if str.contains(allUp, "MGC")
        out := "MGC"
    else if str.contains(allUp, "SIL")
        out := "SIL"
    else if str.contains(allUp, "MHG")
        out := "MHG"
    else if str.contains(allUp, "NQ")
        out := "NQ"
    else if str.contains(allUp, "ES")
        out := "ES"
    else if str.contains(allUp, "YM")
        out := "YM"
    else if str.contains(allUp, "GC")
        out := "GC"
    else if str.contains(allUp, "SI")
        out := "SI"
    else if str.contains(allUp, "PL")
        out := "PL"
    else if str.contains(allUp, "HG")
        out := "HG"
    else if str.contains(allUp, "CL")
        out := "CL"

    string s = f_ssmt_sym_ticker(sym)
    s := str.replace_all(s, "}", "")
    s := str.replace_all(s, "{", "")
    s := str.replace_all(s, "\"", "")
    s := str.replace_all(s, "'", "")
    s := str.replace_all(s, "=", "")
    s := str.trim(s)
    string up = str.upper(s)
    if out == ""
        out := up
        if str.length(up) >= 3
            string r3 = str.substring(up, 0, 3)
            if r3 == "MES" or r3 == "MNQ" or r3 == "MYM" or r3 == "MGC" or r3 == "SIL" or r3 == "MHG"
                out := r3
        if out == up and str.length(up) >= 2
            string r2 = str.substring(up, 0, 2)
            if r2 == "ES" or r2 == "NQ" or r2 == "YM" or r2 == "GC" or r2 == "SI" or r2 == "PL" or r2 == "HG" or r2 == "CL"
                out := r2
    out

f_ssmt_match_chart_contract_size(string inputSym, string chartSym) =>
    string out = inputSym
    string chartT = f_ssmt_sym_ticker(chartSym)

    bool chartMicro = str.contains(chartT, "MES") or str.contains(chartT, "MNQ") or str.contains(chartT, "MYM")
    bool chartMini = (str.contains(chartT, "ES") and not str.contains(chartT, "MES")) or (str.contains(chartT, "NQ") and not str.contains(chartT, "MNQ")) or (str.contains(chartT, "YM") and not str.contains(chartT, "MYM"))

    if chartMicro or chartMini
        string up = str.upper(inputSym)
        string[] parts = str.split(up, ":")
        string ex = array.size(parts) > 1 ? array.get(parts, 0) : ""
        string tk = array.size(parts) > 1 ? array.get(parts, array.size(parts) - 1) : up

        bool tkMicro = str.length(tk) >= 3 and (str.substring(tk, 0, 3) == "MES" or str.substring(tk, 0, 3) == "MNQ" or str.substring(tk, 0, 3) == "MYM")
        bool tkMini = str.length(tk) >= 2 and (str.substring(tk, 0, 2) == "ES" or str.substring(tk, 0, 2) == "NQ" or str.substring(tk, 0, 2) == "YM")

        if chartMicro and tkMini
            string root2 = str.substring(tk, 0, 2)
            string suffix2 = str.substring(tk, 2, str.length(tk))
            if root2 == "ES"
                tk := "MES" + suffix2
            else if root2 == "NQ"
                tk := "MNQ" + suffix2
            else if root2 == "YM"
                tk := "MYM" + suffix2
        else if chartMini and tkMicro
            string root3 = str.substring(tk, 0, 3)
            string suffix3 = str.substring(tk, 3, str.length(tk))
            if root3 == "MES"
                tk := "ES" + suffix3
            else if root3 == "MNQ"
                tk := "NQ" + suffix3
            else if root3 == "MYM"
                tk := "YM" + suffix3

        out := str.length(ex) > 0 ? ex + ":" + tk : tk
    out

f_ssmt_apply_futures_modifiers(string sym, bool useBackadjust) =>
    string out = sym
    if syminfo.type == "futures"
        bool isContinuous = str.contains(f_ssmt_sym_ticker(sym), "!")
        if isContinuous
            if useBackadjust
                out := ticker.modify(sym, session=syminfo.session, backadjustment=backadjustment.on)
            else
                out := ticker.modify(sym, session=syminfo.session, backadjustment=backadjustment.off)
        else
            out := ticker.modify(sym, session=syminfo.session)
        string outTrim = str.trim(out)
        if str.length(outTrim) == 0 or outTrim == "="
            out := sym
    out

f_ssmt_chart_backadjust_state(bool fallback) =>
    bool out = fallback
    if syminfo.type == "futures"
        string tid = str.upper(syminfo.tickerid)
        bool hasOn = str.contains(tid, "BACKADJUSTMENT.ON") or str.contains(tid, "BACKADJUSTMENT=ON") or str.contains(tid, "BACKADJUSTMENT=TRUE") or str.contains(tid, "\"BACKADJUSTMENT\":\"ON\"") or str.contains(tid, "\"BACKADJUSTMENT\":\"TRUE\"")
        bool hasOff = str.contains(tid, "BACKADJUSTMENT.OFF") or str.contains(tid, "BACKADJUSTMENT=OFF") or str.contains(tid, "BACKADJUSTMENT=FALSE") or str.contains(tid, "\"BACKADJUSTMENT\":\"OFF\"") or str.contains(tid, "\"BACKADJUSTMENT\":\"FALSE\"")
        if hasOn and not hasOff
            out := true
        else if hasOff and not hasOn
            out := false
    out

f_set_or_make_line(line ln, int x1, float y1, int x2, float y2, color c, int w) =>
    line out = ln
    if na(out)
        out := line.new(x1, y1, x2, y2, xloc=xloc.bar_time, extend=extend.none, style=line.style_solid, width=w, color=c)
    else
        line.set_xy1(out, x1, y1)
        line.set_xy2(out, x2, y2)
        line.set_extend(out, extend.none)
        line.set_style(out, line.style_solid)
        line.set_width(out, w)
        line.set_color(out, c)
    out

f_set_or_make_hline_w_style(line ln, int x1, int x2, float y, color c, int w, string styleStr) =>
    line out = ln
    if na(out)
        out := line.new(x1, y, x2, y, xloc=xloc.bar_time, extend=extend.none, style=lineStyleFrom(styleStr), width=w, color=c)
    else
        line.set_xy1(out, x1, y)
        line.set_xy2(out, x2, y)
        line.set_extend(out, extend.none)
        line.set_style(out, lineStyleFrom(styleStr))
        line.set_width(out, w)
        line.set_color(out, c)
    out

f_ssmt_hist_push(line ln, label lb, line[] lines, label[] labels, int maxN) =>
    if not na(ln) or not na(lb)
        if maxN <= 0
            if not na(ln)
                line.delete(ln)
            if not na(lb)
                label.delete(lb)
        else
            array.push(lines, ln)
            array.push(labels, lb)
            while array.size(lines) > maxN
                line oldLn = array.shift(lines)
                if not na(oldLn)
                    line.delete(oldLn)
                label oldLb = array.shift(labels)
                if not na(oldLb)
                    label.delete(oldLb)
    0

f_ssmt_hist_trim(line[] lines, label[] labels, int maxN) =>
    while array.size(lines) > maxN
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    while array.size(labels) > maxN
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

f_ssmt_hist_clear(line[] lines, label[] labels) =>
    while array.size(lines) > 0
        line oldLn = array.shift(lines)
        if not na(oldLn)
            line.delete(oldLn)
    while array.size(labels) > 0
        label oldLb = array.shift(labels)
        if not na(oldLb)
            label.delete(oldLb)
    0

f_kill(line ln) =>
    line out = na
    if not na(ln)
        line.delete(ln)
    out
