crossesTs(int ts) =>
    bool havePrev = bar_index > 0 and not na(time[1])
    bool overlap = time <= ts and time_close > ts
    bool gapCross = havePrev and time >= ts and time[1] < ts
    overlap or gapCross

openAtTsLTF(int ts) =>
    float out = na

    float tfSec = timeframe.in_seconds()
    bool subMinute = timeframe.isintraday and not na(tfSec) and tfSec < 60
    bool aboveMinute = timeframe.isintraday and not na(tfSec) and tfSec > 60
    bool useLower = subMinute or aboveMinute
    string ltf = subMinute ? "1S" : "1"

    if useLower
        int[] tArr = request.security_lower_tf(syminfo.tickerid, ltf, time)
        float[] oArr = request.security_lower_tf(syminfo.tickerid, ltf, open)

        int n = array.size(tArr)
        if n > 0
            int i = 0
            bool found = false
            while i < n and not found
                if array.get(tArr, i) == ts
                    out := array.get(oArr, i)
                    found := true
                i += 1
    else if time == ts
        out := open

    if na(out) and crossesTs(ts)
        out := open

    out

newsLineEndTs(int ts) =>
    int yy = year(ts, NY_TZ)
    int mm = month(ts, NY_TZ)
    int dd = dayofmonth(ts, NY_TZ)
    timestamp(NY_TZ, yy, mm, dd + 1, 0, 0)

hinLineStyleFrom(string s) =>
    switch s
        "Solid" => line.style_solid
        "Dashed" => line.style_dashed
        => line.style_dotted

sameNYDay(int tsA, int tsB) =>
    year(tsA, NY_TZ) == year(tsB, NY_TZ) and month(tsA, NY_TZ) == month(tsB, NY_TZ) and dayofmonth(tsA, NY_TZ) == dayofmonth(tsB, NY_TZ)

defaultHINEndTs(int ts) =>
    int yy = year(ts, NY_TZ)
    int mm = month(ts, NY_TZ)
    int dd = dayofmonth(ts, NY_TZ)
    int hh = hour(ts, NY_TZ)
    int fallback = hh < 12 ? timestamp(NY_TZ, yy, mm, dd, 12, 0) : timestamp(NY_TZ, yy, mm, dd, 16, 30)
    fallback > ts ? fallback : newsLineEndTs(ts)

nextHINEndTs(ffUtil.News[] N, int idx) =>
    ffUtil.News event = N.get(idx)
    int out = defaultHINEndTs(event.tmst)
    int n = N.size()
    if idx < n - 1
        for i = idx + 1 to n - 1
            ffUtil.News nextEvent = N.get(i)
            bool uniqueNextTs = nextEvent.tmst > event.tmst
            if uniqueNextTs
                if sameNYDay(event.tmst, nextEvent.tmst)
                    out := nextEvent.tmst
                break
    out

hinLabelText(int ts, string ttl) =>
    str.tostring(hour(ts, NY_TZ)) + ":" + str.tostring(minute(ts, NY_TZ), "00") + " " + ttl

findHINLineIndex(int[] starts, int ts) =>
    int out = -1
    int n = array.size(starts)
    if n > 0
        for i = 0 to n - 1
            if array.get(starts, i) == ts
                out := i
                break
    out

clearHINLines(line[] lines, label[] labels, int[] starts) =>
    int nL = array.size(lines)
    if nL > 0
        for i = 0 to nL - 1
            line li = array.get(lines, i)
            if not na(li)
                li.delete()

    int nB = array.size(labels)
    if nB > 0
        for i = 0 to nB - 1
            label lb = array.get(labels, i)
            if not na(lb)
                lb.delete()

    array.clear(lines)
    array.clear(labels)
    array.clear(starts)
    0

syncHINStyles(line[] lines, label[] labels, color labelColor, string lineStyle, color lineColor, int lineWidth) =>
    int nL = array.size(lines)
    if nL > 0
        for i = 0 to nL - 1
            line li = array.get(lines, i)
            if not na(li)
                li.set_style(hinLineStyleFrom(lineStyle))
                li.set_color(lineColor)
                li.set_width(lineWidth)

    int nB = array.size(labels)
    if nB > 0
        for i = 0 to nB - 1
            label lb = array.get(labels, i)
            if not na(lb)
                lb.set_textcolor(labelColor)
    0

removeHINLinesBefore(line[] lines, label[] labels, int[] starts, int cutoffTs) =>
    int i = array.size(starts) - 1
    while i >= 0
        if array.get(starts, i) < cutoffTs
            line li = array.get(lines, i)
            label lb = array.get(labels, i)
            if not na(li)
                li.delete()
            if not na(lb)
                lb.delete()
            array.remove(lines, i)
            array.remove(labels, i)
            array.remove(starts, i)
        i -= 1
    0

titleOverrideFor(int ts, string cur, string ttl) =>
    string out = ttl
    int mo = month(ts, NY_TZ)
    int dd = dayofmonth(ts, NY_TZ)
    int hh = hour(ts, NY_TZ)
    int mi = minute(ts, NY_TZ)
    int n = array.size(overrideMonths)
    if n > 0
        for i = 0 to n - 1
            bool sameTime = mo == array.get(overrideMonths, i) and dd == array.get(overrideDays, i) and hh == array.get(overrideHours, i) and mi == array.get(overrideMinutes, i)
            bool sameEvent = cur == array.get(overrideCurrencies, i) and ttl == array.get(overrideFromTitles, i)
            if sameTime and sameEvent
                out := array.get(overrideToTitles, i)
                break
    out

applyTitleOverrides(ffUtil.News[] N) =>
    int n = N.size()
    if n > 0
        for i = 0 to n - 1
            ffUtil.News event = N.get(i)
            string overrideTitle = titleOverrideFor(event.tmst, event.cur, event.ttl)
            if overrideTitle != event.ttl
                event.ttl := overrideTitle
                N.set(i, event)
